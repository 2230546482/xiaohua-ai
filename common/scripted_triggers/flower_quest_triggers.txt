#1937小徽章考核、黑市悬赏、半年军令状与黑市契约达标判定触发器

#1.1937小徽章考核：军用工厂扩建/现役陆军师扩编/战时攻占敌方地区（三选一）
flower_quest_st_mil_met = {
	check_variable = { flower_st_goal_mil > 0 }
	NOT = { check_variable = { num_of_military_factories < flower_st_goal_mil } }
}

flower_quest_st_div_met = {
	check_variable = { flower_st_goal_divs > 0 }
	NOT = { check_variable = { num_divisions < flower_st_goal_divs } }
}

flower_quest_st_cap_met = {
	check_variable = { flower_st_goal_caps > 0 }
	NOT = { check_variable = { flower_st_war_caps < flower_st_goal_caps } }
}

flower_quest_st_ready = {
	has_country_flag = flower_st_task_assigned
	NOT = { has_country_flag = flower_st_claimed }
	OR = {
		has_country_flag = flower_st_task_done
		flower_quest_st_mil_met = yes
		flower_quest_st_div_met = yes
		flower_quest_st_cap_met = yes
	}
}

#2.黑市动态悬赏：库存军火交付/接单后战时攻占地区/现役空军扩编（三选一，完成后冷却180天）
flower_quest_bounty_order_met = {
	OR = {
		AND = {
			check_variable = { flower_bounty_order_tier = 1 }
			has_equipment = { infantry_equipment > 999 }
			has_equipment = { artillery_equipment > 99 }
		}
		AND = {
			check_variable = { flower_bounty_order_tier = 2 }
			has_equipment = { infantry_equipment > 1999 }
			has_equipment = { artillery_equipment > 199 }
		}
		AND = {
			check_variable = { flower_bounty_order_tier > 2 }
			has_equipment = { infantry_equipment > 3499 }
			has_equipment = { artillery_equipment > 349 }
		}
	}
}

flower_quest_bounty_cap_met = {
	check_variable = { flower_bounty_goal_caps > 0 }
	NOT = { check_variable = { flower_bounty_war_caps < flower_bounty_goal_caps } }
}

flower_quest_bounty_air_met = {
	check_variable = { flower_bounty_goal_air > 0 }
	OR = {
		AND = { check_variable = { flower_bounty_goal_air = 200 } has_deployed_air_force_size = { size > 199 } }
		AND = { check_variable = { flower_bounty_goal_air = 400 } has_deployed_air_force_size = { size > 399 } }
		AND = { check_variable = { flower_bounty_goal_air = 700 } has_deployed_air_force_size = { size > 699 } }
		AND = { check_variable = { flower_bounty_goal_air = 1000 } has_deployed_air_force_size = { size > 999 } }
		AND = { check_variable = { flower_bounty_goal_air = 1500 } has_deployed_air_force_size = { size > 1499 } }
		AND = { check_variable = { flower_bounty_goal_air = 2200 } has_deployed_air_force_size = { size > 2199 } }
		AND = { check_variable = { flower_bounty_goal_air = 3000 } has_deployed_air_force_size = { size > 2999 } }
		AND = { check_variable = { flower_bounty_goal_air > 3000 } has_deployed_air_force_size = { size > 3999 } }
	}
}

flower_quest_bounty_ready = {
	has_country_flag = flower_bounty_active
	OR = {
		flower_quest_bounty_cap_met = yes
		flower_quest_bounty_air_met = yes
		flower_quest_bounty_order_met = yes
	}
}

#3.半年扩张军令状：控制地区净增/攻占敌方工业地区或迫敌停止抵抗（二选一，完成后冷却180天）
