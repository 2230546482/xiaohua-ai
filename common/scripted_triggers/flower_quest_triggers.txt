flower_quest_st_fac_met = {
	check_variable = { flower_st_goal_fac > 0 }
	NOT = { check_variable = { num_of_factories < flower_st_goal_fac } }
}

flower_quest_st_state_met = {
	check_variable = { flower_st_goal_states > 0 }
	NOT = { check_variable = { num_controlled_states < flower_st_goal_states } }
}

flower_quest_st_ready = {
	has_country_flag = flower_st_task_assigned
	NOT = { has_country_flag = flower_st_claimed }
	OR = {
		has_country_flag = flower_st_task_done
		flower_quest_st_fac_met = yes
		flower_quest_st_state_met = yes
		flower_st_xp_met = yes
	}
}
