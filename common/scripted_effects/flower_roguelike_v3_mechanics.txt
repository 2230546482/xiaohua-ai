flower_ai_recalc_black_market_cost = {
	set_variable = { flower_black_market_cost = AHES_Net_income }
	if = {
		limit = { check_variable = { flower_black_market_cost > 0 } }
		multiply_variable = { flower_black_market_cost = 6 }
	}
	else = {
		set_variable = { flower_black_market_cost = AHES_Total_income }
		multiply_variable = { flower_black_market_cost = 3 }
	}
	round_variable = flower_black_market_cost
	if = {
		limit = { check_variable = { flower_black_market_cost < 250 } }
		set_variable = { flower_black_market_cost = 250 }
	}
	if = {
		limit = { has_country_flag = flower_roguelike_gambler_active }
		multiply_variable = { flower_black_market_cost = 0.8 }
		round_variable = flower_black_market_cost
		if = {
			limit = { check_variable = { flower_black_market_cost < 200 } }
			set_variable = { flower_black_market_cost = 200 }
		}
	}
	#预估高利贷所需额度（供事件文案实时显示）
	if = {
		limit = { check_variable = { ubs_save < flower_black_market_cost } }
		set_variable = { flower_bm_loan_preview = flower_black_market_cost }
		if = {
			limit = { check_variable = { ubs_save > 0 } }
			subtract_from_variable = { flower_bm_loan_preview = ubs_save }
		}
		multiply_variable = { flower_bm_loan_preview = 1.2 }
		round_variable = flower_bm_loan_preview
	}
	else = {
		set_variable = { flower_bm_loan_preview = 0 }
	}
}

flower_ai_pay_black_market_cost = {
	flower_ai_recalc_black_market_cost = yes
	if = {
		limit = { NOT = { check_variable = { ubs_save < flower_black_market_cost } } }
		subtract_from_variable = { ubs_save = flower_black_market_cost }
	}
	else = {
		set_temp_variable = { temp_bm_shortfall = flower_black_market_cost }
		if = {
			limit = { check_variable = { ubs_save > 0 } }
			subtract_from_variable = { temp_bm_shortfall = ubs_save }
			set_variable = { ubs_save = 0 }
		}
		multiply_variable = { temp_bm_shortfall = 1.2 }
		round_variable = temp_bm_shortfall
		add_to_variable = { ubs_load = temp_bm_shortfall }
	}
	flower_ai_sync_hcp_ui = yes
}

#内部选点评分器：在目标敌国作用域调用
#入参：flower_dvl_slot_mode (1=战略中枢/重兵矛头, 2=军工心脏/大机场, 3=前线补给/次高价值重兵)
#      flower_dvl_desp_mode (0=常规开战咏唱, 1=濒死绝唱)
#出参：flower_dvl_best_st_id (最高价值受控州ID，无可用州为0；调用方负责标记flower_dvl_target_tmp)
flower_ai_dvl_pick_best_state = {
	set_variable = { flower_dvl_best_st_score = -9999 }
	set_variable = { flower_dvl_best_st_id = 0 }
	every_controlled_state = {
		limit = {
			NOT = { has_state_flag = flower_dvl_target_tmp }
			impassable = no
		}
		set_temp_variable = { st_score = 1 }
		#1. 军用工厂权重（落点2重点定点拔除军工心脏）
		set_temp_variable = { t_af = building_level@arms_factory }
		if = {
			limit = { check_variable = { PREV.flower_dvl_slot_mode = 2 } }
			multiply_temp_variable = { t_af = 20 }
		}
		else = {
			multiply_temp_variable = { t_af = 14 }
		}
		add_to_temp_variable = { st_score = t_af }
		#2. 民用工厂权重
		set_temp_variable = { t_ic = building_level@industrial_complex }
		multiply_temp_variable = { t_ic = 8 }
		add_to_temp_variable = { st_score = t_ic }
		#3. 空军基地权重（落点2重点摧毁大机场）
		set_temp_variable = { t_ab = building_level@air_base }
		if = {
			limit = { check_variable = { PREV.flower_dvl_slot_mode = 2 } }
			multiply_temp_variable = { t_ab = 9 }
		}
		else = {
			multiply_temp_variable = { t_ab = 5 }
		}
		add_to_temp_variable = { st_score = t_ab }
		#4. 战略胜利点价值
		set_temp_variable = { t_sv = state_strategic_value }
		multiply_temp_variable = { t_sv = 1.2 }
		add_to_temp_variable = { st_score = t_sv }
		#5. 核反应堆与补给枢纽
		if = {
			limit = { nuclear_reactor > 0 }
			add_to_temp_variable = { st_score = 35 }
			if = {
				limit = { check_variable = { PREV.flower_dvl_slot_mode = 2 } }
				add_to_temp_variable = { st_score = 20 }
			}
		}
		if = {
			limit = { supply_node > 0 }
			add_to_temp_variable = { st_score = 22 }
			if = {
				limit = { check_variable = { PREV.flower_dvl_slot_mode = 3 } }
				add_to_temp_variable = { st_score = 22 }
			}
		}
		#6. 本土核心、直连区、欧洲战区与避开无人孤岛
		if = {
			limit = { is_core_of = PREV }
			add_to_temp_variable = { st_score = 25 }
		}
		if = {
			limit = { is_in_home_area = yes }
			add_to_temp_variable = { st_score = 20 }
		}
		if = {
			limit = { is_on_continent = europe }
			add_to_temp_variable = { st_score = 15 }
		}
		if = {
			limit = { is_island_state = yes }
			add_to_temp_variable = { st_score = -35 }
		}
		#7. 首都加权（常规落点1优先首都，但若前线出现超大重兵集团则允许前线反超；濒死时淡化空壳首都）
		if = {
			limit = { is_capital = yes }
			if = {
				limit = {
					check_variable = { PREV.flower_dvl_slot_mode = 1 }
					check_variable = { PREV.flower_dvl_desp_mode = 0 }
				}
				add_to_temp_variable = { st_score = 45 }
			}
			else = {
				add_to_temp_variable = { st_score = 15 }
			}
		}
		#8. 实时重兵集群侦测（核爆优先砸向敌方兵力最密集的集团军驻地）
		if = {
			limit = { controller = { divisions_in_state = { state = PREV size > 2 } } }
			add_to_temp_variable = { st_score = 30 }
			if = {
				limit = { check_variable = { PREV.flower_dvl_desp_mode > 0 } }
				add_to_temp_variable = { st_score = 15 }
			}
		}
		if = {
			limit = { controller = { divisions_in_state = { state = PREV size > 6 } } }
			add_to_temp_variable = { st_score = 45 }
			if = {
				limit = { check_variable = { PREV.flower_dvl_desp_mode > 0 } }
				add_to_temp_variable = { st_score = 25 }
			}
		}
		if = {
			limit = { controller = { divisions_in_state = { state = PREV size > 12 } } }
			add_to_temp_variable = { st_score = 70 }
			if = {
				limit = { check_variable = { PREV.flower_dvl_desp_mode > 0 } }
				add_to_temp_variable = { st_score = 45 }
			}
		}
		#9. 交战第一线与二线纵深地缘加权（优先摧毁压在魔导国边境线上的进攻矛头）
		if = {
			limit = {
				any_neighbor_state = {
					controller = {
						OR = {
							tag = DVL
							is_in_faction_with = DVL
						}
					}
				}
			}
			if = {
				limit = { check_variable = { PREV.flower_dvl_desp_mode > 0 } }
				add_to_temp_variable = { st_score = 110 }
			}
			else = {
				add_to_temp_variable = { st_score = 55 }
			}
			if = {
				limit = { check_variable = { PREV.flower_dvl_slot_mode = 3 } }
				add_to_temp_variable = { st_score = 25 }
			}
		}
		else_if = {
			limit = {
				any_neighbor_state = {
					any_neighbor_state = {
						controller = {
							OR = {
								tag = DVL
								is_in_faction_with = DVL
							}
						}
					}
				}
			}
			add_to_temp_variable = { st_score = 18 }
		}
		if = {
			limit = { check_variable = { st_score > PREV.flower_dvl_best_st_score } }
			set_temp_variable = { t_cur_st_id = THIS.id }
			PREV = {
				set_variable = { flower_dvl_best_st_score = st_score }
				set_variable = { flower_dvl_best_st_id = t_cur_st_id }
			}
		}
	}
	clear_variable = flower_dvl_best_st_score
}

flower_ai_check_ai_catchup = {
	if = {
		limit = {
			has_global_flag = flower_anomaly_1_active
			is_ai = yes
			is_major = yes
			NOT = { tag = ZDW }
			NOT = { tag = WL1 }
			NOT = { has_country_flag = flower_ai_catchup_cd }
		}
		set_temp_variable = { temp_ai_need_cards = flower_roguelike_cards_count }
		add_to_temp_variable = { temp_ai_need_cards = 2 }
		if = {
			limit = {
				any_country = {
					is_ai = no
					OR = {
						has_war_with = ROOT
						is_neighbor_of = ROOT
					}
					check_variable = { flower_roguelike_cards_count > temp_ai_need_cards }
				}
			}
			set_country_flag = { flag = flower_ai_catchup_cd days = 180 value = 1 }
			flower_ai_open_roguelike_draft = yes
		}
	}
}
