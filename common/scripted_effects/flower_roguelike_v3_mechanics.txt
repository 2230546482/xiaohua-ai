#一、黑市6个月净收入契约（支持现金支付与1.2倍高利贷负债）

flower_ai_recalc_black_market_cost = {
	set_variable = { flower_black_market_cost = AHES_Net_income }
	if = {
		limit = { check_variable = { flower_black_market_cost > 0 } }
		multiply_variable = { flower_black_market_cost = 6 }
	}
	else = {
		set_variable = { flower_black_market_cost = AHES_Total_income }
		multiply_variable = { flower_black_market_cost = 3 }
	}
	round_variable = flower_black_market_cost
	#按年份阶段动态抬升黑市契约保底价（1936-37: 250 / 1938-39: 500 / 1940+: 750）
	set_temp_variable = { temp_bm_floor = 250 }
	if = {
		limit = { date > 1939.12.31 }
		set_temp_variable = { temp_bm_floor = 750 }
	}
	else_if = {
		limit = { date > 1937.12.31 }
		set_temp_variable = { temp_bm_floor = 500 }
	}
	if = {
		limit = { check_variable = { flower_black_market_cost < temp_bm_floor } }
		set_variable = { flower_black_market_cost = temp_bm_floor }
	}
	if = {
		limit = {
			OR = {
				has_country_flag = flower_roguelike_gambler_active
				has_idea = flower_st_econ_vip
			}
		}
		multiply_variable = { flower_black_market_cost = 0.75 }
		round_variable = flower_black_market_cost
		if = {
			limit = { check_variable = { flower_black_market_cost < 200 } }
			set_variable = { flower_black_market_cost = 200 }
		}
	}
	#预估高利贷所需额度（供事件文案实时显示）
	if = {
		limit = { check_variable = { ubs_save < flower_black_market_cost } }
		set_variable = { flower_bm_loan_preview = flower_black_market_cost }
		if = {
			limit = { check_variable = { ubs_save > 0 } }
			subtract_from_variable = { flower_bm_loan_preview = ubs_save }
		}
		multiply_variable = { flower_bm_loan_preview = 1.2 }
		round_variable = flower_bm_loan_preview
	}
	else = {
		set_variable = { flower_bm_loan_preview = 0 }
	}
}

flower_ai_pay_black_market_cost = {
	flower_ai_recalc_black_market_cost = yes
	if = {
		limit = { NOT = { check_variable = { ubs_save < flower_black_market_cost } } }
		subtract_from_variable = { ubs_save = flower_black_market_cost }
	}
	else = {
		set_temp_variable = { temp_bm_shortfall = flower_black_market_cost }
		if = {
			limit = { check_variable = { ubs_save > 0 } }
			subtract_from_variable = { temp_bm_shortfall = ubs_save }
			set_variable = { ubs_save = 0 }
		}
		multiply_temp_variable = { temp_bm_shortfall = 1.2 }
		round_temp_variable = temp_bm_shortfall
		add_to_variable = { ubs_load = temp_bm_shortfall }
	}
	#词条51《战争之王的供货单》：真正签约扣费后返还200储备
	if = {
		limit = { has_country_flag = flower_card_owned_51 }
		add_to_variable = { ubs_save = 200 }
	}
	flower_ai_sync_hcp_ui = yes
}

			#锁定落点3：前线补给枢纽或次高价值重兵集群州
			set_variable = { flower_dvl_slot_mode = 3 }
			flower_ai_dvl_pick_best_state = yes
			if = {
				limit = { check_variable = { flower_dvl_best_st_id > 0 } }
				set_variable = { flower_dvl_warn_st_3 = flower_dvl_best_st_id }
				var:flower_dvl_warn_st_3 = {
					set_state_flag = flower_dvl_target_tmp
					save_event_target_as = flower_dvl_target_st_3
				}
			}
			clear_variable = flower_dvl_slot_mode
			clear_variable = flower_dvl_best_st_id
			every_controlled_state = {
				limit = { has_state_flag = flower_dvl_target_tmp }
				clr_state_flag = flower_dvl_target_tmp
			}
			#目标国不足3州时复用落点1显示
			if = {
				limit = {
					check_variable = { flower_dvl_warn_st_2 = 0 }
					check_variable = { flower_dvl_warn_st_1 > 0 }
				}
				event_target:flower_dvl_target_st_1 = { save_event_target_as = flower_dvl_target_st_2 }
			}
			if = {
				limit = {
					check_variable = { flower_dvl_warn_st_3 = 0 }
					check_variable = { flower_dvl_warn_st_1 > 0 }
				}
				event_target:flower_dvl_target_st_1 = { save_event_target_as = flower_dvl_target_st_3 }
			}
			#延迟1天触发预警弹窗并在确认交战态稳定后激活21天倒计时任务，杜绝宣战同刻误取消
			if = {
				limit = { check_variable = { flower_dvl_desp_mode > 0 } }
				country_event = { id = FLWAI.44 days = 1 }
			}
			else = {
				set_country_flag = { flag = flower_gutou_super_magic_nuked days = 730 value = 1 }
				country_event = { id = FLWAI.43 days = 1 }
			}
		}
	}
}
