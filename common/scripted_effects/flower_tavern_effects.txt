flower_tavern_calc_dynamic_goals = {
	#维度1：按年份持续递增的时间系数（1936=1.00至1943+=3.10）
	set_variable = { flower_quest_time_mult = 1.00 }
	set_variable = { flower_quest_diff_pct = 100 }
	set_variable = { flower_quest_diff_stars = 1 }
	if = {
		limit = { date > 1942.12.31 }
		set_variable = { flower_quest_time_mult = 3.10 }
		set_variable = { flower_quest_diff_pct = 310 }
		set_variable = { flower_quest_diff_stars = 5 }
	}
	else_if = {
		limit = { date > 1941.12.31 }
		set_variable = { flower_quest_time_mult = 2.65 }
		set_variable = { flower_quest_diff_pct = 265 }
		set_variable = { flower_quest_diff_stars = 5 }
	}
	else_if = {
		limit = { date > 1940.12.31 }
		set_variable = { flower_quest_time_mult = 2.25 }
		set_variable = { flower_quest_diff_pct = 225 }
		set_variable = { flower_quest_diff_stars = 4 }
	}
	else_if = {
		limit = { date > 1939.12.31 }
		set_variable = { flower_quest_time_mult = 1.90 }
		set_variable = { flower_quest_diff_pct = 190 }
		set_variable = { flower_quest_diff_stars = 4 }
	}
	else_if = {
		limit = { date > 1938.12.31 }
		set_variable = { flower_quest_time_mult = 1.60 }
		set_variable = { flower_quest_diff_pct = 160 }
		set_variable = { flower_quest_diff_stars = 3 }
	}
	else_if = {
		limit = { date > 1937.12.31 }
		set_variable = { flower_quest_time_mult = 1.35 }
		set_variable = { flower_quest_diff_pct = 135 }
		set_variable = { flower_quest_diff_stars = 2 }
	}
	else_if = {
		limit = { date > 1936.12.31 }
		set_variable = { flower_quest_time_mult = 1.15 }
		set_variable = { flower_quest_diff_pct = 115 }
		set_variable = { flower_quest_diff_stars = 2 }
	}

	#维度2：四项差异化考核指标（军用工厂/民用工厂/陆军师/战时攻占地区/攻占工业地区/空军扩编/军火订单/储备净增长）
	set_temp_variable = { t_fac_base = num_of_factories }
	multiply_temp_variable = { t_fac_base = 0.12 }
	add_to_temp_variable = { t_fac_base = 3 }
	multiply_temp_variable = { t_fac_base = flower_quest_time_mult }
	set_variable = { flower_dyn_fac_delta = t_fac_base }
	round_variable = flower_dyn_fac_delta
	if = { limit = { check_variable = { flower_dyn_fac_delta < 3 } } set_variable = { flower_dyn_fac_delta = 3 } }
	if = { limit = { check_variable = { flower_dyn_fac_delta > 120 } } set_variable = { flower_dyn_fac_delta = 120 } }

	#军用工厂扩建增量（用于1937小徽章考核）
	set_temp_variable = { t_mil_base = num_of_military_factories }
	multiply_temp_variable = { t_mil_base = 0.15 }
	add_to_temp_variable = { t_mil_base = 3 }
	multiply_temp_variable = { t_mil_base = flower_quest_time_mult }
	set_variable = { flower_dyn_mil_delta = t_mil_base }
	round_variable = flower_dyn_mil_delta
	if = { limit = { check_variable = { flower_dyn_mil_delta < 3 } } set_variable = { flower_dyn_mil_delta = 3 } }
	if = { limit = { check_variable = { flower_dyn_mil_delta > 45 } } set_variable = { flower_dyn_mil_delta = 45 } }

	#民用工厂扩建增量（用于黑市净入账契约）
	set_temp_variable = { t_civ_base = num_of_civilian_factories }
	multiply_temp_variable = { t_civ_base = 0.14 }
	add_to_temp_variable = { t_civ_base = 3 }
	multiply_temp_variable = { t_civ_base = flower_quest_time_mult }
	set_variable = { flower_dyn_civ_delta = t_civ_base }
	round_variable = flower_dyn_civ_delta
	if = { limit = { check_variable = { flower_dyn_civ_delta < 3 } } set_variable = { flower_dyn_civ_delta = 3 } }
	if = { limit = { check_variable = { flower_dyn_civ_delta > 45 } } set_variable = { flower_dyn_civ_delta = 45 } }

	#现役陆军师扩编增量（用于1937小徽章考核）
	set_temp_variable = { t_div_base = num_divisions }
	multiply_temp_variable = { t_div_base = 0.25 }
	add_to_temp_variable = { t_div_base = 6 }
	multiply_temp_variable = { t_div_base = flower_quest_time_mult }
	set_variable = { flower_dyn_div_delta = t_div_base }
	round_variable = flower_dyn_div_delta
	if = { limit = { check_variable = { flower_dyn_div_delta < 6 } } set_variable = { flower_dyn_div_delta = 6 } }
	if = { limit = { check_variable = { flower_dyn_div_delta > 60 } } set_variable = { flower_dyn_div_delta = 60 } }

	#控制地区净增（用于半年扩张军令状）
	set_temp_variable = { t_st_base = num_controlled_states }
	multiply_temp_variable = { t_st_base = 0.10 }
	if = { limit = { check_variable = { t_st_base > 10 } } set_temp_variable = { t_st_base = 10 } }
	add_to_temp_variable = { t_st_base = 2 }
	multiply_temp_variable = { t_st_base = flower_quest_time_mult }
	set_variable = { flower_dyn_state_delta = t_st_base }
	round_variable = flower_dyn_state_delta
	if = { limit = { check_variable = { flower_dyn_state_delta < 3 } } set_variable = { flower_dyn_state_delta = 3 } }
	if = { limit = { check_variable = { flower_dyn_state_delta > 35 } } set_variable = { flower_dyn_state_delta = 35 } }

	#战时实占敌方地区数（用于1937小徽章考核与黑市动态悬赏）
	set_temp_variable = { t_cap_base = flower_quest_time_mult }
	multiply_temp_variable = { t_cap_base = 2 }
	add_to_temp_variable = { t_cap_base = 2 }
	set_variable = { flower_dyn_cap_goal = t_cap_base }
	round_variable = flower_dyn_cap_goal
	if = { limit = { check_variable = { flower_dyn_cap_goal < 3 } } set_variable = { flower_dyn_cap_goal = 3 } }
	if = { limit = { check_variable = { flower_dyn_cap_goal > 10 } } set_variable = { flower_dyn_cap_goal = 10 } }

	#战时攻占敌方带工厂工业重镇数（用于半年扩张军令状）
	set_temp_variable = { t_ind_base = flower_quest_time_mult }
	add_to_temp_variable = { t_ind_base = 1 }
	set_variable = { flower_dyn_ind_cap_goal = t_ind_base }
	round_variable = flower_dyn_ind_cap_goal
	if = { limit = { check_variable = { flower_dyn_ind_cap_goal < 2 } } set_variable = { flower_dyn_ind_cap_goal = 2 } }
	if = { limit = { check_variable = { flower_dyn_ind_cap_goal > 6 } } set_variable = { flower_dyn_ind_cap_goal = 6 } }

	#黑市契约签约后国库储备净增长目标（50整档取整）
	set_temp_variable = { t_hcp_step = flower_quest_time_mult }
	multiply_temp_variable = { t_hcp_step = 7 }
	round_temp_variable = t_hcp_step
	if = { limit = { check_variable = { t_hcp_step < 7 } } set_temp_variable = { t_hcp_step = 7 } }
	if = { limit = { check_variable = { t_hcp_step > 24 } } set_temp_variable = { t_hcp_step = 24 } }
	set_variable = { flower_dyn_bm_hcp_delta = t_hcp_step }
	multiply_variable = { flower_dyn_bm_hcp_delta = 50 }

	#黑市动态悬赏军火订单数量（按年份与工厂总数分级，交付时扣减库存）
	set_variable = { flower_dyn_order_tier = 1 }
	set_variable = { flower_dyn_order_inf = 1000 }
	set_variable = { flower_dyn_order_art = 100 }
	if = {
		limit = { OR = { date > 1940.12.31 num_of_factories > 119 } }
		set_variable = { flower_dyn_order_tier = 3 }
		set_variable = { flower_dyn_order_inf = 3500 }
		set_variable = { flower_dyn_order_art = 350 }
	}
	else_if = {
		limit = { OR = { date > 1938.12.31 num_of_factories > 59 } }
		set_variable = { flower_dyn_order_tier = 2 }
		set_variable = { flower_dyn_order_inf = 2000 }
		set_variable = { flower_dyn_order_art = 200 }
	}

	#黑市动态悬赏现役空军扩编目标（高于当前已部署空军规模）
	set_variable = { flower_dyn_air_next_goal = 200 }
	if = { limit = { has_deployed_air_force_size = { size > 2599 } } set_variable = { flower_dyn_air_next_goal = 4000 } }
	else_if = { limit = { has_deployed_air_force_size = { size > 1899 } } set_variable = { flower_dyn_air_next_goal = 3000 } }
	else_if = { limit = { has_deployed_air_force_size = { size > 1299 } } set_variable = { flower_dyn_air_next_goal = 2200 } }
	else_if = { limit = { has_deployed_air_force_size = { size > 849 } } set_variable = { flower_dyn_air_next_goal = 1500 } }
	else_if = { limit = { has_deployed_air_force_size = { size > 549 } } set_variable = { flower_dyn_air_next_goal = 1000 } }
	else_if = { limit = { has_deployed_air_force_size = { size > 299 } } set_variable = { flower_dyn_air_next_goal = 700 } }
	else_if = { limit = { has_deployed_air_force_size = { size > 149 } } set_variable = { flower_dyn_air_next_goal = 400 } }

	#预估下一轮各任务赏金（随轮次递增，用于任务页签即时预览，不奖励陆军经验）
	set_temp_variable = { t_b_rnd = flower_bounty_round }
	if = { limit = { check_variable = { t_b_rnd > 5 } } set_temp_variable = { t_b_rnd = 5 } }
	set_variable = { flower_bounty_reward_hcp = t_b_rnd }
	multiply_variable = { flower_bounty_reward_hcp = 100 }
	add_to_variable = { flower_bounty_reward_hcp = 350 }
	set_variable = { flower_bounty_disp_round = flower_bounty_round }
	add_to_variable = { flower_bounty_disp_round = 1 }

	set_temp_variable = { t_w_rnd = flower_wager_round }
	if = { limit = { check_variable = { t_w_rnd > 5 } } set_temp_variable = { t_w_rnd = 5 } }
	set_variable = { flower_wager_reward_hcp = t_w_rnd }
	multiply_variable = { flower_wager_reward_hcp = 100 }
	add_to_variable = { flower_wager_reward_hcp = 400 }
	set_variable = { flower_wager_disp_round = flower_wager_round }
	add_to_variable = { flower_wager_disp_round = 1 }

	set_temp_variable = { t_bm_rnd = flower_bm_round }
	if = { limit = { check_variable = { t_bm_rnd > 6 } } set_temp_variable = { t_bm_rnd = 6 } }
	set_variable = { flower_bm_reward_hcp = t_bm_rnd }
	multiply_variable = { flower_bm_reward_hcp = 50 }
	add_to_variable = { flower_bm_reward_hcp = 200 }
	set_variable = { flower_bm_disp_round = flower_bm_round }
	add_to_variable = { flower_bm_disp_round = 1 }

	#计算1939大徽章基准价格（按工厂规模、贵宾折扣徽章与已持有常驻战术卡减免计算）
	set_temp_variable = { t_size = 1 }
	if = { limit = { num_of_factories > 69 } set_temp_variable = { t_size = 3 } }
	else_if = { limit = { num_of_factories > 29 } set_temp_variable = { t_size = 2 } }
	set_variable = { flower_lt_std_cost = 400 }
	set_variable = { flower_lt_cheap_cost = 200 }
	if = {
		limit = { check_variable = { t_size = 2 } }
		set_variable = { flower_lt_std_cost = 500 }
		set_variable = { flower_lt_cheap_cost = 250 }

#依次弹出队列中下一张待替换的战术卡
flower_tavern_pop_next_pending_card = {
	if = {
		limit = {
			is_ai = no
			check_variable = { flower_pending_new_card_id = 0 }
			check_variable = { flower_pending_cards_queue^num > 0 }
		}
		set_temp_variable = { temp_grant_card_id = flower_pending_cards_queue^0 }
		remove_from_array = {
			array = flower_pending_cards_queue
			index = 0
		}
		flower_tavern_acquire_card_with_cap = yes
	}
}

#战术卡获取与5槽上限结算流程（已有同名卡直接合并升级，满5槽时进入替换选择）
flower_tavern_acquire_card_with_cap = {
	set_temp_variable = { temp_card_id = temp_grant_card_id }
	flower_tavern_check_is_slot_card = yes
	if = {
		limit = {
			check_variable = { temp_is_slot_card = 1 }
			is_in_array = {
				array = flower_owned_cards_array
				value = temp_grant_card_id
			}
		}
		#同名常驻卡直接原地升阶为对子或三连金色卡
		flower_tavern_grant_card_by_id = yes
	}
	else_if = {
		limit = {
			check_variable = { temp_is_slot_card = 1 }
			check_variable = { flower_owned_cards_array^num > 4 }
		}
		if = {
			limit = { is_ai = no }
			if = {
				limit = { check_variable = { flower_pending_new_card_id > 0 } }
				add_to_array = { flower_pending_cards_queue = temp_grant_card_id }
			}
			else = {
				set_variable = { flower_pending_new_card_id = temp_grant_card_id }
				set_variable = { flower_replace_slot_1 = flower_owned_cards_array^0 }
				set_variable = { flower_replace_slot_2 = flower_owned_cards_array^1 }
				set_variable = { flower_replace_slot_3 = flower_owned_cards_array^2 }
				set_variable = { flower_replace_slot_4 = flower_owned_cards_array^3 }
				set_variable = { flower_replace_slot_5 = flower_owned_cards_array^4 }
				country_event = { id = FLWTV.20 }
			}
		}
		else = {
			flower_tavern_ai_make_room_for_card = yes
			flower_tavern_grant_card_by_id = yes
		}
	}
	else = {
		flower_tavern_grant_card_by_id = yes
	}
}
#击败主要敌国截获战术卡与战利品陈列馆拓印小徽章
flower_tavern_trigger_steal_from_defeated = {
	if = {
		limit = {
			OR = {
				has_global_flag = flower_anomaly_1_active
				has_global_flag = flower_anomaly_4_active
			}
			NOT = { tag = ZDW }
			NOT = { tag = WL1 }
			NOT = { has_country_flag = flower_steal_card_cd }
		}
		set_country_flag = { flag = flower_steal_card_cd days = 90 value = 1 }
		set_variable = { flower_steal_opt_1 = 0 }
		set_variable = { flower_steal_opt_2 = 0 }
		set_variable = { flower_steal_opt_3 = 0 }
		var:temp_steal_victim = {
			for_each_loop = {
				array = flower_owned_cards_array
				value = v_card
				if = {
					limit = {
						check_variable = { v_card > 0 }
						PREV = { flower_can_upgrade_v_card = yes }
					}
					if = {
						limit = { PREV = { check_variable = { flower_steal_opt_1 = 0 } } }
						PREV = { set_variable = { flower_steal_opt_1 = v_card } }
					}
					else_if = {
						limit = {
							PREV = {
								check_variable = { flower_steal_opt_2 = 0 }
								NOT = { check_variable = { flower_steal_opt_1 = v_card } }
