#辅助：将临时变量 t_raw_xp 对齐到12档离散陆军经验表，输出到 t_snapped_xp
flower_tavern_snap_xp_goal = {
	set_temp_variable = { t_snapped_xp = 40 }
	if = { limit = { check_variable = { t_raw_xp > 250 } } set_temp_variable = { t_snapped_xp = 300 } }
	else_if = { limit = { check_variable = { t_raw_xp > 200 } } set_temp_variable = { t_snapped_xp = 250 } }
	else_if = { limit = { check_variable = { t_raw_xp > 180 } } set_temp_variable = { t_snapped_xp = 200 } }
	else_if = { limit = { check_variable = { t_raw_xp > 150 } } set_temp_variable = { t_snapped_xp = 180 } }
	else_if = { limit = { check_variable = { t_raw_xp > 120 } } set_temp_variable = { t_snapped_xp = 150 } }
	else_if = { limit = { check_variable = { t_raw_xp > 100 } } set_temp_variable = { t_snapped_xp = 120 } }
	else_if = { limit = { check_variable = { t_raw_xp > 90 } } set_temp_variable = { t_snapped_xp = 100 } }
	else_if = { limit = { check_variable = { t_raw_xp > 75 } } set_temp_variable = { t_snapped_xp = 90 } }
	else_if = { limit = { check_variable = { t_raw_xp > 60 } } set_temp_variable = { t_snapped_xp = 75 } }
	else_if = { limit = { check_variable = { t_raw_xp > 50 } } set_temp_variable = { t_snapped_xp = 60 } }
	else_if = { limit = { check_variable = { t_raw_xp > 40 } } set_temp_variable = { t_snapped_xp = 50 } }
}

flower_tavern_calc_dynamic_goals = {
	#维度1：按年份持续递增的时间系数（1936=1.00至1943+=3.10，不再于1940年封顶）
	set_variable = { flower_quest_time_mult = 1.00 }
	set_variable = { flower_quest_diff_pct = 100 }
	set_variable = { flower_quest_diff_stars = 1 }
	if = {
		limit = { date > 1942.12.31 }
		set_variable = { flower_quest_time_mult = 3.10 }
		set_variable = { flower_quest_diff_pct = 310 }
		set_variable = { flower_quest_diff_stars = 5 }
	}
	else_if = {
		limit = { date > 1941.12.31 }
		set_variable = { flower_quest_time_mult = 2.65 }
		set_variable = { flower_quest_diff_pct = 265 }
		set_variable = { flower_quest_diff_stars = 5 }
	}
	else_if = {
		limit = { date > 1940.12.31 }
		set_variable = { flower_quest_time_mult = 2.25 }
		set_variable = { flower_quest_diff_pct = 225 }
		set_variable = { flower_quest_diff_stars = 4 }
	}
	else_if = {
		limit = { date > 1939.12.31 }
		set_variable = { flower_quest_time_mult = 1.90 }
		set_variable = { flower_quest_diff_pct = 190 }
		set_variable = { flower_quest_diff_stars = 4 }
	}
	else_if = {
		limit = { date > 1938.12.31 }
		set_variable = { flower_quest_time_mult = 1.60 }
		set_variable = { flower_quest_diff_pct = 160 }
		set_variable = { flower_quest_diff_stars = 3 }
	}
	else_if = {
		limit = { date > 1937.12.31 }
		set_variable = { flower_quest_time_mult = 1.35 }
		set_variable = { flower_quest_diff_pct = 135 }
		set_variable = { flower_quest_diff_stars = 2 }
	}
	else_if = {
		limit = { date > 1936.12.31 }
		set_variable = { flower_quest_time_mult = 1.15 }
		set_variable = { flower_quest_diff_pct = 115 }
		set_variable = { flower_quest_diff_stars = 2 }
	}

	#维度2：按当前总工厂数与控制地区数比例缩放（取消70厂封顶）
	set_temp_variable = { t_fac_base = num_of_factories }
	multiply_temp_variable = { t_fac_base = 0.12 }
	add_to_temp_variable = { t_fac_base = 3 }
	multiply_temp_variable = { t_fac_base = flower_quest_time_mult }
	set_variable = { flower_dyn_fac_delta = t_fac_base }
	round_variable = flower_dyn_fac_delta
	if = { limit = { check_variable = { flower_dyn_fac_delta < 3 } } set_variable = { flower_dyn_fac_delta = 3 } }
	if = { limit = { check_variable = { flower_dyn_fac_delta > 120 } } set_variable = { flower_dyn_fac_delta = 120 } }

	set_variable = { flower_dyn_mil_delta = flower_dyn_fac_delta }
	multiply_variable = { flower_dyn_mil_delta = 0.65 }
	round_variable = flower_dyn_mil_delta
	if = { limit = { check_variable = { flower_dyn_mil_delta < 2 } } set_variable = { flower_dyn_mil_delta = 2 } }

	set_temp_variable = { t_st_base = num_controlled_states }
	multiply_temp_variable = { t_st_base = 0.10 }
	if = { limit = { check_variable = { t_st_base > 10 } } set_temp_variable = { t_st_base = 10 } }
	add_to_temp_variable = { t_st_base = 2 }
	multiply_temp_variable = { t_st_base = flower_quest_time_mult }
	set_variable = { flower_dyn_state_delta = t_st_base }
	round_variable = flower_dyn_state_delta
	if = { limit = { check_variable = { flower_dyn_state_delta < 2 } } set_variable = { flower_dyn_state_delta = 2 } }
	if = { limit = { check_variable = { flower_dyn_state_delta > 35 } } set_variable = { flower_dyn_state_delta = 35 } }

	set_temp_variable = { t_raw_xp = num_of_factories }
	multiply_temp_variable = { t_raw_xp = 0.35 }
	add_to_temp_variable = { t_raw_xp = 35 }
	multiply_temp_variable = { t_raw_xp = flower_quest_time_mult }
	flower_tavern_snap_xp_goal = yes
	set_variable = { flower_dyn_xp_goal = t_snapped_xp }

	#预估下一轮各任务赏金（随轮次递增，用于任务页签即时预览）
	set_temp_variable = { t_b_rnd = flower_bounty_round }
	if = { limit = { check_variable = { t_b_rnd > 5 } } set_temp_variable = { t_b_rnd = 5 } }
	set_variable = { flower_bounty_reward_hcp = t_b_rnd }
	multiply_variable = { flower_bounty_reward_hcp = 100 }
	add_to_variable = { flower_bounty_reward_hcp = 350 }
	set_temp_variable = { t_b_xpr = flower_bounty_round }
	if = { limit = { check_variable = { t_b_xpr > 4 } } set_temp_variable = { t_b_xpr = 4 } }
	set_variable = { flower_bounty_reward_xp = t_b_xpr }
	multiply_variable = { flower_bounty_reward_xp = 10 }
	add_to_variable = { flower_bounty_reward_xp = 35 }
	set_variable = { flower_bounty_disp_round = flower_bounty_round }
	add_to_variable = { flower_bounty_disp_round = 1 }

	set_temp_variable = { t_w_rnd = flower_wager_round }
	if = { limit = { check_variable = { t_w_rnd > 5 } } set_temp_variable = { t_w_rnd = 5 } }
	set_variable = { flower_wager_reward_hcp = t_w_rnd }
	multiply_variable = { flower_wager_reward_hcp = 100 }
	add_to_variable = { flower_wager_reward_hcp = 400 }
	set_variable = { flower_wager_disp_round = flower_wager_round }
	add_to_variable = { flower_wager_disp_round = 1 }

	set_temp_variable = { t_bm_rnd = flower_bm_round }
	if = { limit = { check_variable = { t_bm_rnd > 6 } } set_temp_variable = { t_bm_rnd = 6 } }
	set_variable = { flower_bm_reward_hcp = t_bm_rnd }
	multiply_variable = { flower_bm_reward_hcp = 50 }
	add_to_variable = { flower_bm_reward_hcp = 200 }
	set_variable = { flower_bm_disp_round = flower_bm_round }
	add_to_variable = { flower_bm_disp_round = 1 }

	#小徽章买断价与大徽章基准价（绑定体量、七五折徽章与常驻战术卡抵扣）
	set_temp_variable = { t_size = 1 }
	if = { limit = { num_of_factories > 69 } set_temp_variable = { t_size = 3 } }
	else_if = { limit = { num_of_factories > 29 } set_temp_variable = { t_size = 2 } }
	set_variable = { flower_st_buyout_cost = 250 }
	set_variable = { flower_lt_std_cost = 400 }
	set_variable = { flower_lt_cheap_cost = 200 }
	if = {
		limit = { check_variable = { t_size = 2 } }
		set_variable = { flower_st_buyout_cost = 350 }
		set_variable = { flower_lt_std_cost = 500 }
		set_variable = { flower_lt_cheap_cost = 250 }
	}
	else_if = {
		limit = { check_variable = { t_size = 3 } }
		set_variable = { flower_st_buyout_cost = 500 }
		set_variable = { flower_lt_std_cost = 750 }
		set_variable = { flower_lt_cheap_cost = 350 }
	}
	if = {
		limit = { has_idea = flower_st_econ_vip }
		multiply_variable = { flower_st_buyout_cost = 0.75 }
		multiply_variable = { flower_lt_std_cost = 0.75 }
		multiply_variable = { flower_lt_cheap_cost = 0.75 }
		round_variable = flower_st_buyout_cost
		round_variable = flower_lt_std_cost
		round_variable = flower_lt_cheap_cost
	}
	#每持有1张常驻战术卡，1939大徽章标准价立减50储备（上限减200储备，保底150）
	set_temp_variable = { t_card_disc = flower_owned_cards_array^num }
	if = { limit = { check_variable = { t_card_disc > 4 } } set_temp_variable = { t_card_disc = 4 } }
	if = { limit = { check_variable = { t_card_disc < 0 } } set_temp_variable = { t_card_disc = 0 } }
	set_variable = { flower_lt_card_discount = t_card_disc }
	multiply_variable = { flower_lt_card_discount = 50 }
	subtract_from_variable = { flower_lt_std_cost = flower_lt_card_discount }
	if = { limit = { check_variable = { flower_lt_std_cost < 150 } } set_variable = { flower_lt_std_cost = 150 } }
	set_temp_variable = { t_cheap_disc = t_card_disc }
	multiply_temp_variable = { t_cheap_disc = 25 }
	subtract_from_variable = { flower_lt_cheap_cost = t_cheap_disc }
	if = { limit = { check_variable = { flower_lt_cheap_cost < 100 } } set_variable = { flower_lt_cheap_cost = 100 } }
}

	set_variable = { flower_pivot_saved_stage = 1 }

		limit = {
			OR = {
				has_idea = flower_corp_stage_3_idea
				has_idea = flower_arty_stage_3_idea
				has_idea = flower_tank_stage_3_idea
				has_idea = flower_air_stage_3_idea
				has_idea = flower_wave_stage_3_idea
				has_idea = flower_nightingale_stage_3_idea
				has_idea = flower_guerrilla_stage_3_idea
				has_idea = flower_undead_stage_3_idea
				has_idea = flower_magic_stage_3_idea
				has_idea = flower_steampunk_stage_3_idea
				AND = { has_country_flag = flower_roguelike_slayer_ai check_variable = { flower_slayer_stacks > 4 } }
				AND = { has_country_flag = flower_roguelike_gambler_active has_country_flag = flower_mystery_egg_hatched_flag }
			}
		}
		set_variable = { flower_pivot_saved_stage = 3 }
	}
	else_if = {
		limit = {
			OR = {
				has_idea = flower_corp_stage_2_idea
				has_idea = flower_arty_stage_2_idea
				has_idea = flower_tank_stage_2_idea

		}
		if = {
			limit = { is_ai = no }

				limit = { check_variable = { flower_pending_new_card_id > 0 } }

				set_variable = { flower_pending_new_card_id = temp_grant_card_id }

#===============================================================================
#八、月度节点调度与AI酒馆全权运营（1937小饰品、1939大饰品、1939+转型、1941终局桂冠）
#===============================================================================
flower_tavern_monthly_tick = {
	if = {
		limit = {
			OR = {
				has_global_flag = flower_anomaly_1_active
				has_global_flag = flower_anomaly_4_active
			}
			OR = {
				is_ai = no
				is_major = yes
			}
			NOT = { tag = ZDW }
			NOT = { tag = WL1 }
		}
		flower_tavern_recalc_trinket_dynamics = yes
		flower_tavern_check_golden_synthesis = yes

		#1. 1937年6月：发放小徽章动态考核（记录起始工厂与地区基线用于进度条计算）
		if = {
			limit = {
				date > 1937.5.31
				NOT = { has_country_flag = flower_st_task_assigned }
				NOT = { has_country_flag = flower_st_claimed }
			}
			set_country_flag = flower_st_task_assigned
			flower_tavern_calc_dynamic_goals = yes
			set_variable = { flower_st_start_fac = num_of_factories }
			set_variable = { flower_st_goal_fac = num_of_factories }
			add_to_variable = { flower_st_goal_fac = flower_dyn_fac_delta }
			set_variable = { flower_st_start_states = num_controlled_states }
			set_variable = { flower_st_goal_states = num_controlled_states }
			add_to_variable = { flower_st_goal_states = flower_dyn_state_delta }
			set_variable = { flower_st_goal_xp = flower_dyn_xp_goal }
			if = {
				limit = { is_ai = no }
				set_country_flag = flower_badge_hub_alert
				scoped_sound_effect = "alert_pick_new_idea_sound"
				flower_badge_hub_sync_state = yes
			}
		}

		#2. 检测1937小徽章考核是否达成 -> 玩家通过GUI就地4选1领取，AI走后台事件
		if = {
			limit = {
				has_country_flag = flower_st_task_assigned
				NOT = { has_country_flag = flower_st_claimed }
				NOT = { has_country_flag = flower_st_task_done }
				flower_quest_st_ready = yes
			}
			set_country_flag = flower_st_task_done
			flower_tavern_roll_small_trinkets = yes
			if = {
				limit = { is_ai = no }
				set_country_flag = flower_badge_hub_alert
				scoped_sound_effect = "flower_sfx_combo_awakening"
				flower_badge_hub_sync_state = yes
			}
			else = {
				country_event = { id = FLWTV.2 }
			}
		}
		#2.5 若玩家有任意悬赏/军令状/黑市契约已达标可领赏，点亮悬浮窗红点提醒
		if = {
			limit = {
				is_ai = no
				flower_quest_any_claimable = yes
			}
			set_country_flag = flower_badge_hub_alert
		}

		#3. AI若至1938年仍未完成小徽章考核：付真金储备买断，实在没钱则进候选拿0费诅咒槽
		if = {
			limit = {
				is_ai = yes
				date > 1938.2.1
				has_country_flag = flower_st_task_assigned
				NOT = { has_country_flag = flower_st_claimed }
				NOT = { has_country_flag = flower_st_ai_buyout_tried }
			}
			set_country_flag = flower_st_ai_buyout_tried
			flower_tavern_calc_dynamic_goals = yes
			if = {
				limit = { NOT = { check_variable = { ubs_save < flower_st_buyout_cost } } }
				subtract_from_variable = { ubs_save = flower_st_buyout_cost }
				flower_ai_sync_hcp_ui = yes
				set_country_flag = flower_st_task_done
			}
			flower_tavern_roll_small_trinkets = yes
			country_event = { id = FLWTV.2 }
		}

		#4. 1939年6月：开启1939大徽章4槽候选（玩家通过GUI就地选购并亮红点提醒，AI走后台事件）
		if = {
			limit = {
				date > 1939.5.31
				NOT = { has_country_flag = flower_lt_shop_opened }
				NOT = { has_country_flag = flower_lt_claimed }
			}
			set_country_flag = flower_lt_shop_opened
			flower_tavern_roll_large_trinkets = yes
			if = {
				limit = { is_ai = no }
				set_country_flag = flower_badge_hub_alert
				scoped_sound_effect = "alert_pick_new_idea_sound"
				flower_badge_hub_sync_state = yes
			}
			else = {
				country_event = { id = FLWTV.3 }
			}
		}
		else_if = {
			limit = {
				is_ai = yes
				has_country_flag = flower_lt_shop_deferred
				NOT = { has_country_flag = flower_lt_claimed }
				OR = {
					NOT = { check_variable = { ubs_save < flower_lt_cheap_cost } }
					has_political_power > 124.99
				}
			}
			clr_country_flag = flower_lt_shop_deferred
			flower_tavern_calc_dynamic_goals = yes
			country_event = { id = FLWTV.3 }
		}

		#4.5 1941年1月：玩家超大徽章开放，点亮右上角悬浮窗红点提醒
		if = {
			limit = {
				is_ai = no
				date > 1940.12.31
				NOT = { has_country_flag = flower_crown_claimed }
				NOT = { has_country_flag = flower_crown_gui_notified }
			}
			set_country_flag = flower_crown_gui_notified
			set_country_flag = flower_badge_hub_alert
			scoped_sound_effect = "alert_pick_new_idea_sound"
			flower_badge_hub_sync_state = yes
		}

		if = {
			limit = {
				is_ai = yes
				date > 1938.12.31
				has_war = yes
				has_country_flag = flower_roguelike_corp_ai
				num_of_factories > 60
				NOT = { has_country_flag = flower_ai_pivoted_once }

						has_political_power > 149
						check_variable = { ubs_save > 249 }
					}
				}
			}
			set_country_flag = flower_ai_pivoted_once
			if = {
				limit = { NOT = { has_idea = flower_lt_econ_pivot } }
				add_political_power = -150
				add_to_variable = { ubs_save = -250 }
				flower_ai_sync_hcp_ui = yes
			}
			flower_tavern_clear_current_archetype = yes
			set_variable = { flower_tavern_tmp_arch = 0 }
			random_list = {
				10 = {
					modifier = { factor = 0 AND = { has_global_flag = flower_tavern_pool_rolled NOT = { has_global_flag = flower_pool_arch_1 } } }
					set_variable = { flower_tavern_tmp_arch = 1 }
				}
				15 = {
					modifier = { factor = 0 AND = { has_global_flag = flower_tavern_pool_rolled NOT = { has_global_flag = flower_pool_arch_3 } } }
					modifier = { factor = 1.8 num_of_military_factories > 35 }
					set_variable = { flower_tavern_tmp_arch = 3 }
				}
				15 = {
					modifier = { factor = 0 AND = { has_global_flag = flower_tavern_pool_rolled NOT = { has_global_flag = flower_pool_arch_4 } } }
					modifier = { factor = 2.0 num_of_factories > 90 }
					modifier = { factor = 0.4 num_of_military_factories < 30 }
					set_variable = { flower_tavern_tmp_arch = 4 }
				}
				12 = {
					modifier = { factor = 0 AND = { has_global_flag = flower_tavern_pool_rolled NOT = { has_global_flag = flower_pool_arch_5 } } }
					modifier = { factor = 2.0 has_deployed_air_force_size = { size > 500 } }
					modifier = { factor = 0.2 has_deployed_air_force_size = { size < 150 } }
					set_variable = { flower_tavern_tmp_arch = 5 }
				}

					modifier = { factor = 1.6 has_manpower > 800000 }
					set_variable = { flower_tavern_tmp_arch = 9 }
				}

					modifier = { factor = 0 AND = { has_global_flag = flower_tavern_pool_rolled NOT = { has_global_flag = flower_pool_arch_10 } } }
					set_variable = { flower_tavern_tmp_arch = 10 }
