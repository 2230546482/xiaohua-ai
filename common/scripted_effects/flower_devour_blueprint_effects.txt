flower_unlock_bp_t3_den_next = {
	if = {
		limit = { NOT = { has_tech = steamtank2 } }
		set_technology = { steamtank2 = 1 popup = no }
	}
	else_if = {
		limit = { NOT = { has_tech = steamtank3 } }
		set_technology = { steamtank3 = 1 popup = no }
	}
	else_if = {
		limit = { NOT = { has_tech = steamtank4 } }
		set_technology = { steamtank4 = 1 popup = no }
	}
	flower_ai_sync_steampunk_upgrades = yes
}
