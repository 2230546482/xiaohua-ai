#为单个国家补挂当前已激活的7~12号全局畸变初始精神与前置科技
flower_ai_sync_country_anomalies_v2 = {
	if = {
		limit = {
			NOT = { tag = ZDW }
			NOT = { tag = WL1 }
		}
		if = {
			limit = {
				has_global_flag = flower_anomaly_7_active
				NOT = { has_idea = flower_anom7_base_idea }
			}
			add_ideas = flower_anom7_base_idea
		}
		if = {
			limit = {
				has_global_flag = flower_anomaly_8_active
				NOT = { has_idea = flower_anom8_base_idea }
			}
			add_ideas = flower_anom8_base_idea
		}
		if = {
			limit = {
				has_global_flag = flower_anomaly_9_active
				NOT = { has_idea = flower_anom9_base_idea }
			}
			add_ideas = flower_anom9_base_idea
		}
		if = {
			limit = {
				has_global_flag = flower_anomaly_10_active
				NOT = {
					has_idea = flower_anom10_calm_idea
					has_idea = flower_anom10_war_idea
				}
			}
			if = {
				limit = { has_war = yes }
				add_ideas = flower_anom10_war_idea
			}
			else = {
				add_ideas = flower_anom10_calm_idea
			}
		}
		if = {
			limit = {
				has_global_flag = flower_anomaly_11_active
			}
			if = {
				limit = { NOT = { has_idea = flower_anom11_base_idea } }
				add_ideas = flower_anom11_base_idea
			}
			if = {
				limit = { NOT = { has_country_flag = flower_anom11_techs_given } }
				set_country_flag = flower_anom11_techs_given
				hidden_effect = {
					set_technology = {
						paratroopers = 1
						marines = 1
						popup = no
					}
					if = {
						limit = { has_dlc = "Man the Guns" }
						set_technology = { mtg_transport = 1 popup = no }
					}
					else = {
						set_technology = { transport = 1 popup = no }
					}
				}
			}
		}
		if = {
			limit = {
				has_global_flag = flower_anomaly_12_active
				NOT = { has_idea = flower_anom12_base_idea }
			}
			add_ideas = flower_anom12_base_idea
		}
	}
}

#投票结算瞬间统一激活7~12号全局畸变

	if = {
		limit = {
			OR = {
				has_global_flag = flower_anomaly_7_active
				has_global_flag = flower_anomaly_8_active

				has_global_flag = flower_anomaly_11_active

			}
		}

	}
}

#畸变7：单国实时重算【骄兵必败】与三阶【哀兵必胜】（月度巡检与丢核心省瞬间双端调用）
flower_ai_anom7_refresh_tiers = {
	if = {
		limit = {
			has_global_flag = flower_anomaly_7_active
			NOT = { tag = ZDW }
			NOT = { tag = WL1 }
		}
		if = {
			limit = { has_war = no }
			if = { limit = { has_idea = flower_anom7_arrogant_idea } remove_ideas = flower_anom7_arrogant_idea }
			if = { limit = { has_idea = flower_anom7_desperate_1_idea } remove_ideas = flower_anom7_desperate_1_idea }
			if = { limit = { has_idea = flower_anom7_desperate_2_idea } remove_ideas = flower_anom7_desperate_2_idea }
			if = { limit = { has_idea = flower_anom7_desperate_3_idea } remove_ideas = flower_anom7_desperate_3_idea }
		}
		else_if = {
			limit = { surrender_progress > 0.60 }
			if = { limit = { has_idea = flower_anom7_arrogant_idea } remove_ideas = flower_anom7_arrogant_idea }
			if = { limit = { has_idea = flower_anom7_desperate_1_idea } remove_ideas = flower_anom7_desperate_1_idea }
			if = { limit = { has_idea = flower_anom7_desperate_2_idea } remove_ideas = flower_anom7_desperate_2_idea }
			if = { limit = { NOT = { has_idea = flower_anom7_desperate_3_idea } } add_ideas = flower_anom7_desperate_3_idea }
		}
		else_if = {
			limit = { surrender_progress > 0.40 }
			if = { limit = { has_idea = flower_anom7_arrogant_idea } remove_ideas = flower_anom7_arrogant_idea }
			if = { limit = { has_idea = flower_anom7_desperate_1_idea } remove_ideas = flower_anom7_desperate_1_idea }
			if = { limit = { has_idea = flower_anom7_desperate_3_idea } remove_ideas = flower_anom7_desperate_3_idea }
			if = { limit = { NOT = { has_idea = flower_anom7_desperate_2_idea } } add_ideas = flower_anom7_desperate_2_idea }
		}
		else_if = {
			limit = { surrender_progress > 0.20 }
			if = { limit = { has_idea = flower_anom7_arrogant_idea } remove_ideas = flower_anom7_arrogant_idea }
			if = { limit = { has_idea = flower_anom7_desperate_2_idea } remove_ideas = flower_anom7_desperate_2_idea }
			if = { limit = { has_idea = flower_anom7_desperate_3_idea } remove_ideas = flower_anom7_desperate_3_idea }
			if = { limit = { NOT = { has_idea = flower_anom7_desperate_1_idea } } add_ideas = flower_anom7_desperate_1_idea }
		}
		else_if = {
			limit = { surrender_progress < 0.05 }
			set_temp_variable = { anom7_occ_cnt = 0 }
			every_controlled_state = {
				limit = { NOT = { is_owned_by = PREV } }
				add_to_temp_variable = { anom7_occ_cnt = 1 }
			}
			if = { limit = { has_idea = flower_anom7_desperate_1_idea } remove_ideas = flower_anom7_desperate_1_idea }
			if = { limit = { has_idea = flower_anom7_desperate_2_idea } remove_ideas = flower_anom7_desperate_2_idea }
			if = { limit = { has_idea = flower_anom7_desperate_3_idea } remove_ideas = flower_anom7_desperate_3_idea }
			if = {
				limit = {
					check_variable = { anom7_occ_cnt > 5 }
					NOT = { has_idea = flower_anom7_arrogant_idea }
				}
				add_ideas = flower_anom7_arrogant_idea
			}
			else_if = {
				limit = {
					check_variable = { anom7_occ_cnt < 6 }
					has_idea = flower_anom7_arrogant_idea
				}
				remove_ideas = flower_anom7_arrogant_idea
			}
		}
		else = {
			if = { limit = { has_idea = flower_anom7_arrogant_idea } remove_ideas = flower_anom7_arrogant_idea }
			if = { limit = { has_idea = flower_anom7_desperate_1_idea } remove_ideas = flower_anom7_desperate_1_idea }
			if = { limit = { has_idea = flower_anom7_desperate_2_idea } remove_ideas = flower_anom7_desperate_2_idea }
			if = { limit = { has_idea = flower_anom7_desperate_3_idea } remove_ideas = flower_anom7_desperate_3_idea }
		}
	}
}

#畸变7-12月度巡检
flower_ai_tick_anomalies_v2_monthly = {
	if = {
		limit = {
			NOT = { tag = ZDW }
			NOT = { tag = WL1 }
			OR = {
				has_global_flag = flower_anomaly_7_active
				has_global_flag = flower_anomaly_8_active
				has_global_flag = flower_anomaly_9_active
				has_global_flag = flower_anomaly_10_active
				has_global_flag = flower_anomaly_11_active
				has_global_flag = flower_anomaly_12_active
			}
		}
		flower_ai_sync_country_anomalies_v2 = yes

		#---畸变7：骄兵必败 / 哀兵必胜 三档实时切换---
		if = {
			limit = { has_global_flag = flower_anomaly_7_active }
			flower_ai_anom7_refresh_tiers = yes
		}

		#---畸变8：AI战时嘴硬反弹---
		if = {
			limit = {
				has_global_flag = flower_anomaly_8_active
				is_ai = yes
				has_war = yes
				NOT = { has_country_flag = flower_anom8_boast_cd }
			}
			set_temp_variable = { anom8_strong_enemy_th = num_of_factories }
			multiply_temp_variable = { anom8_strong_enemy_th = 1.5 }
			if = {
				limit = {
					OR = {
						has_stability < 0.50
						surrender_progress > 0.15
						any_enemy_country = {
							check_variable = { num_of_factories > anom8_strong_enemy_th }
						}
					}
				}
				set_country_flag = { flag = flower_anom8_boast_cd days = 120 value = 1 }
				add_timed_idea = { idea = flower_anom8_boast_idea days = 60 }
			}
		}

		#---畸变9：全局每85天定时开盒全图外卖状态（消除2月导致的29天空窗，无缝衔接）---
		if = {
			limit = {
				has_global_flag = flower_anomaly_9_active
				NOT = { has_global_flag = flower_anom9_box_cd }
			}
			set_global_flag = { flag = flower_anom9_box_cd days = 85 value = 1 }
			flower_ai_anom9_roll_global_takeout = yes
		}

		#---畸变10：和平【从从容容】与战时【扛厂跑路体制】平滑切换---
		if = {
			limit = { has_global_flag = flower_anomaly_10_active }
			if = {
				limit = { has_war = no }
				if = { limit = { has_idea = flower_anom10_war_idea } remove_ideas = flower_anom10_war_idea }
				if = { limit = { NOT = { has_idea = flower_anom10_calm_idea } } add_ideas = flower_anom10_calm_idea }
			}
			else = {
				if = { limit = { has_idea = flower_anom10_calm_idea } remove_ideas = flower_anom10_calm_idea }
				if = { limit = { NOT = { has_idea = flower_anom10_war_idea } } add_ideas = flower_anom10_war_idea }
			}
		}
	}
}

			has_global_flag = flower_anomaly_7_active

			set_state_flag = { flag = flower_anom10_just_lost days = 3 value = 1 }

					}
				}
			}
			FROM = {
				set_country_flag = { flag = flower_anom10_evac_cd days = 60 value = 1 }
			}

				limit = { FROM.FROM = { arms_factory > 0 } }

							}
