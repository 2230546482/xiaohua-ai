#统一接取黑市动态悬赏（三维动态难度叠加轮次倍数，记录进度条基线）
flower_quest_accept_bounty = {
	if = {
		limit = { NOT = { has_country_flag = flower_bounty_active } }
		set_country_flag = flower_bounty_active
		flower_tavern_calc_dynamic_goals = yes
		set_temp_variable = { t_rmult = flower_bounty_round }
		if = { limit = { check_variable = { t_rmult > 5 } } set_temp_variable = { t_rmult = 5 } }
		multiply_temp_variable = { t_rmult = 0.20 }
		add_to_temp_variable = { t_rmult = 1.00 }
		set_temp_variable = { t_dfac = flower_dyn_fac_delta }
		multiply_temp_variable = { t_dfac = t_rmult }
		round_temp_variable = t_dfac
		if = { limit = { check_variable = { t_dfac < 3 } } set_temp_variable = { t_dfac = 3 } }
		set_temp_variable = { t_dst = flower_dyn_state_delta }
		multiply_temp_variable = { t_dst = t_rmult }
		round_temp_variable = t_dst
		if = { limit = { check_variable = { t_dst < 2 } } set_temp_variable = { t_dst = 2 } }
		if = { limit = { check_variable = { t_dst > 40 } } set_temp_variable = { t_dst = 40 } }
		set_temp_variable = { t_raw_xp = flower_dyn_xp_goal }
		multiply_temp_variable = { t_raw_xp = t_rmult }
		flower_tavern_snap_xp_goal = yes
		set_variable = { flower_bounty_start_fac = num_of_factories }
		set_variable = { flower_bounty_goal_fac = num_of_factories }
		add_to_variable = { flower_bounty_goal_fac = t_dfac }
		set_variable = { flower_bounty_start_states = num_controlled_states }
		set_variable = { flower_bounty_goal_states = num_controlled_states }
		add_to_variable = { flower_bounty_goal_states = t_dst }
		set_variable = { flower_bounty_goal_xp = t_snapped_xp }
		country_event = { id = FLWTV.31 days = 180 }
		if = {
			limit = { is_ai = no }
			set_variable = { flower_badge_feedback_id = 7 }
			flower_badge_hub_sync_state = yes
		}
	}
}

		limit = { check_variable = { global.flower_months_elapsed < 1 } }
		if = { limit = { date > 1940.12.31 } set_variable = { global.flower_months_elapsed = 60 } }
		else_if = { limit = { date > 1940.6.1 } set_variable = { global.flower_months_elapsed = 54 } }
		else_if = { limit = { date > 1939.12.31 } set_variable = { global.flower_months_elapsed = 48 } }

flower_badge_hub_buyout_st = {
	if = {
		limit = {
			has_country_flag = flower_st_task_assigned
			NOT = { has_country_flag = flower_st_claimed }
			NOT = { has_country_flag = flower_st_task_done }
			OR = {
				NOT = { check_variable = { ubs_save < flower_st_buyout_cost } }
				has_political_power > 149.99
			}
		}
		flower_tavern_calc_dynamic_goals = yes
		if = {
			limit = { NOT = { check_variable = { ubs_save < flower_st_buyout_cost } } }
			subtract_from_variable = { ubs_save = flower_st_buyout_cost }
			flower_ai_sync_hcp_ui = yes
		}
		else = {
			add_political_power = -150
		}
		set_country_flag = flower_st_task_done
		flower_tavern_roll_small_trinkets = yes
		set_variable = { flower_badge_hub_tab = 0 }
		set_variable = { flower_badge_feedback_id = 6 }
		if = { limit = { is_ai = no } scoped_sound_effect = "flower_sfx_branch_stonks_money" }
		flower_badge_hub_sync_state = yes
	}
}
