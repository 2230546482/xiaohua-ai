#战时攻占敌方地区时更新考核、悬赏与军令状战功计数（单地区冷却60天）
flower_quest_on_state_captured = {
	if = {
		limit = {
			ROOT = { has_war_with = FROM }
			FROM.FROM = { NOT = { has_state_flag = flower_quest_cap_counted } }
		}
		FROM.FROM = { set_state_flag = { flag = flower_quest_cap_counted days = 60 value = 1 } }
		ROOT = {
			if = {
				limit = {
					has_country_flag = flower_st_task_assigned
					NOT = { has_country_flag = flower_st_claimed }
					NOT = { has_country_flag = flower_st_task_done }
				}
				add_to_variable = { flower_st_war_caps = 1 }
			}
			if = {
				limit = { has_country_flag = flower_bounty_active }
				add_to_variable = { flower_bounty_war_caps = 1 }
			}
		}
		if = {
			limit = {
				ROOT = { has_idea = flower_blood_wager_active_idea }
				FROM.FROM = {
					OR = {
						arms_factory > 0
						industrial_complex > 0
					}
				}
			}
			ROOT = { add_to_variable = { flower_wager_ind_caps = 1 } }
		}
	}
}

#迫使非内战敌国停止抵抗时直接完成半年军令状破城战功指标
flower_quest_on_enemy_capitulated = {
	if = {
		limit = {
			FROM = { has_idea = flower_blood_wager_active_idea }
			ROOT = {
				NOT = { original_tag = FROM }
				NOT = { tag = ZDW }
				NOT = { tag = WL1 }
				NOT = { tag = WRQ }
			}
		}
		FROM = {
			if = {
				limit = { check_variable = { flower_wager_goal_ind_caps > 0 } }
				set_variable = { flower_wager_ind_caps = flower_wager_goal_ind_caps }
			}
			else = {
				add_to_variable = { flower_wager_ind_caps = 10 }
			}
		}
	}
}

#接取黑市动态悬赏（战时攻占地区/现役空军扩编/军火订单交付，完成后冷却180天）
flower_quest_accept_bounty = {
	if = {
		limit = {
			NOT = { has_country_flag = flower_bounty_active }
			NOT = { has_country_flag = flower_bounty_cd }
		}
		set_country_flag = flower_bounty_active
		flower_tavern_calc_dynamic_goals = yes
		set_temp_variable = { t_rmult = flower_bounty_round }
		if = { limit = { check_variable = { t_rmult > 5 } } set_temp_variable = { t_rmult = 5 } }
		multiply_temp_variable = { t_rmult = 0.20 }
		add_to_temp_variable = { t_rmult = 1.00 }
		set_temp_variable = { t_caps = flower_dyn_cap_goal }
		multiply_temp_variable = { t_caps = t_rmult }
		round_temp_variable = t_caps
		if = { limit = { check_variable = { t_caps < 3 } } set_temp_variable = { t_caps = 3 } }
		if = { limit = { check_variable = { t_caps > 15 } } set_temp_variable = { t_caps = 15 } }
		set_variable = { flower_bounty_war_caps = 0 }
		set_variable = { flower_bounty_goal_caps = t_caps }
		set_variable = { flower_bounty_goal_air = flower_dyn_air_next_goal }
		set_variable = { flower_bounty_order_tier = flower_dyn_order_tier }
		set_variable = { flower_bounty_order_inf = 1000 }
		set_variable = { flower_bounty_order_art = 100 }
		if = {
			limit = { check_variable = { flower_bounty_order_tier = 2 } }
			set_variable = { flower_bounty_order_inf = 2000 }
			set_variable = { flower_bounty_order_art = 200 }
		}
		else_if = {
			limit = { check_variable = { flower_bounty_order_tier > 2 } }
			set_variable = { flower_bounty_order_inf = 3500 }
			set_variable = { flower_bounty_order_art = 350 }
		}
		set_country_flag = { flag = flower_bounty_expire_lock days = 179 value = 1 }
		country_event = { id = FLWTV.31 days = 180 }
		if = {
			limit = { is_ai = no }
			set_variable = { flower_badge_feedback_id = 7 }
			flower_badge_hub_sync_state = yes
		}
	}
}

#交付黑市动态悬赏并领取奖励（完成后冷却180天，按军火订单达标时扣减库存装备）
flower_quest_claim_bounty = {
	if = {
		limit = { flower_quest_bounty_ready = yes }
		flower_tavern_calc_dynamic_goals = yes
		#若未走战地夺地或空军扩编，则真实交付军火订单库存
		if = {
			limit = {
				NOT = { flower_quest_bounty_cap_met = yes }
				NOT = { flower_quest_bounty_air_met = yes }
				flower_quest_bounty_order_met = yes
			}
			if = {
				limit = { check_variable = { flower_bounty_order_tier = 1 } }
				add_equipment_to_stockpile = { type = infantry_equipment amount = -1000 }
				add_equipment_to_stockpile = { type = artillery_equipment amount = -100 }
			}
			else_if = {
				limit = { check_variable = { flower_bounty_order_tier = 2 } }
				add_equipment_to_stockpile = { type = infantry_equipment amount = -2000 }
				add_equipment_to_stockpile = { type = artillery_equipment amount = -200 }
			}
			else = {
				add_equipment_to_stockpile = { type = infantry_equipment amount = -3500 }
				add_equipment_to_stockpile = { type = artillery_equipment amount = -350 }
			}
		}
		clr_country_flag = flower_bounty_active

		add_to_variable = { ubs_save = flower_bounty_reward_hcp }

			limit = { has_idea = flower_st_econ_vip }
