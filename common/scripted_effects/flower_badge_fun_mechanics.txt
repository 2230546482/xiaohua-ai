flower_badge_on_claim_st = {
	if = {
		limit = { check_variable = { temp_badge_id = 2 } }
		#标准厂房蓝图：本土核心立即落成1座军用工厂
		random_owned_controlled_state = {
			limit = { is_core_of = ROOT }
			add_extra_state_shared_building_slots = 1
			add_building_construction = {
				type = arms_factory
				level = 1
				instant_build = yes
			}
		}
	}
	else_if = {
		limit = { check_variable = { temp_badge_id = 10 } }
		#特种精锐徽章：提拔1名在任将领获突击队员特质并强化进攻防御
		if = {
			limit = { any_army_leader = { is_assigned = yes NOT = { has_trait = commando } } }
			random_army_leader = {
				limit = { is_assigned = yes NOT = { has_trait = commando } }
				add_unit_leader_trait = commando
				add_attack = 2
				add_defense = 1
			}
		}
		else_if = {
			limit = { any_army_leader = { is_assigned = yes } }
			random_army_leader = {
				limit = { is_assigned = yes }
				add_attack = 2
				add_defense = 1
			}
		}
		else = {
			random_army_leader = {
				add_attack = 2
				add_defense = 1
			}
		}
	}
	else_if = {
		limit = { check_variable = { temp_badge_id = 11 } }
		#实验记录本：赠1张工业或电子科研加速卡
		add_tech_bonus = {
			name = flower_blackbox_s1_bonus
			bonus = 1.0
			uses = 1
			category = industry
			category = electronics
		}
	}
	else_if = {
		limit = { check_variable = { temp_badge_id = 15 } }
		#外包基建合同：本土核心立即落成2座民用工厂
		random_owned_controlled_state = {
			limit = { is_core_of = ROOT }
			add_extra_state_shared_building_slots = 2
			add_building_construction = {
				type = industrial_complex
				level = 2
				instant_build = yes
			}
		}
	}
	else_if = {
		limit = { check_variable = { temp_badge_id = 18 } }
		#名将勋章盒：立即提拔1名现役将领进攻与后勤各+2
		if = {
			limit = { any_army_leader = { is_assigned = yes } }
			random_army_leader = {
				limit = { is_assigned = yes }
				add_attack = 2
				add_logistics = 2
			}
		}
		else = {
			random_army_leader = {
				add_attack = 2
				add_logistics = 2
			}
		}
	}
	else_if = {
		limit = { check_variable = { temp_badge_id = 19 } }
		#列车炮牵引调度令：直接交付1门现役列车炮并附赠火炮加速卡
		if = {
			limit = { NOT = { has_tech = railway_gun } }
			set_technology = { railway_gun = 1 popup = no }
		}
		create_railway_gun = {
			equipment = railway_gun_equipment_1
		}
		add_tech_bonus = {
			name = flower_blackbox_s1_bonus
			bonus = 1.0
			uses = 1
			category = artillery
		}
	}
	else_if = {
		limit = { check_variable = { temp_badge_id = 21 } }
		#战时借款凭证：立即发放300储备
		add_to_variable = { ubs_save = 300 }
		flower_ai_sync_hcp_ui = yes
	}
}
