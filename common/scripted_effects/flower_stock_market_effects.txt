#安全兜底：防止强平、平仓或倒贴分红把国家储备扣成负数（缺口自动转入债务）
flower_stk_clamp_negative_cash = {
	if = {
		limit = { check_variable = { ubs_save < 0 } }
		set_temp_variable = { stk_neg_debt = 0 }
		subtract_from_temp_variable = { stk_neg_debt = ubs_save }
		add_to_variable = { ubs_load = stk_neg_debt }
		set_variable = { ubs_save = 0 }
	}
}

#每月初全局跳价、周期轮动、半年分红与爆仓结算
flower_stock_monthly_tick = {
	#若未初始化则安全补全
	if = {
		limit = { check_variable = { global.flower_stk_1_price < 1 } }
		flower_stock_init_market = yes
	}

	#记录历史走势（前月 <- 上月 <- 今月）
	set_variable = { global.flower_stk_1_prev2 = global.flower_stk_1_prev }
	set_variable = { global.flower_stk_1_prev = global.flower_stk_1_price }
	set_variable = { global.flower_stk_2_prev2 = global.flower_stk_2_prev }
	set_variable = { global.flower_stk_2_prev = global.flower_stk_2_price }
	set_variable = { global.flower_stk_3_prev2 = global.flower_stk_3_prev }
	set_variable = { global.flower_stk_3_prev = global.flower_stk_3_price }
	set_variable = { global.flower_stk_4_prev2 = global.flower_stk_4_prev }
	set_variable = { global.flower_stk_4_prev = global.flower_stk_4_price }

	#1. 依据上月周期与世界紧张度，推演本月大盘风向（牛熊具备惯性，也会反转）
	set_temp_variable = { prev_hint = global.flower_stk_hint }
	random_list = {
		#1 = 超级大牛市
		14 = {
			modifier = { factor = 2.0 check_variable = { global.flower_stk_regime = 1 } }
			modifier = { factor = 1.6 check_variable = { global.flower_stk_regime = 2 } }
			modifier = { factor = 2.4 check_variable = { prev_hint = 5 } }
			modifier = { factor = 0.4 check_variable = { prev_hint = 6 } }
			set_variable = { global.flower_stk_regime = 1 }
		}
		#2 = 温和慢牛
		26 = {
			modifier = { factor = 1.7 check_variable = { global.flower_stk_regime = 2 } }

flower_stk_buy_factory_action = {
	if = {
		limit = {
			NOT = { has_country_flag = flower_stk_buy_factory_cd }
			NOT = { check_variable = { ubs_save < 500 } }
		}
		set_country_flag = { flag = flower_stk_buy_factory_cd days = 60 value = 1 }
		if = {
			limit = {
				is_ai = no
				NOT = { has_country_flag = flower_auto_meme_sfx_cd }
			}
			set_country_flag = { flag = flower_auto_meme_sfx_cd days = 50 value = 1 }
			hidden_effect = { scoped_sound_effect = "flower_sfx_branch_stonks_money" }
		}
		subtract_from_variable = { ubs_save = 500 }
		flower_stk_clamp_negative_cash = yes
	flower_ai_sync_hcp_ui = yes
		if = {
			limit = {
				any_owned_state = {
					is_controlled_by = ROOT
					free_building_slots = {
						building = arms_factory
						size > 1
						include_locked = yes
					}
				}
			}
			random_owned_controlled_state = {
				limit = {
					free_building_slots = {
						building = arms_factory
						size > 1
						include_locked = yes
					}
				}
				add_extra_state_shared_building_slots = 2
				add_building_construction = {
					type = arms_factory
					level = 1
					instant_build = yes
				}
				add_building_construction = {
					type = industrial_complex
					level = 1
					instant_build = yes
				}
			}
		}
		else_if = {
			limit = { any_owned_state = { is_controlled_by = ROOT } }
			random_owned_controlled_state = {
				add_extra_state_shared_building_slots = 2
				add_building_construction = {
					type = arms_factory
					level = 1
					instant_build = yes
				}
				add_building_construction = {
					type = industrial_complex
					level = 1
					instant_build = yes
				}
			}
		}
	}
}

			35 = {
				modifier = { factor = 2.0 OR = { has_government = fascism has_war = yes } }

			}
			40 = {
				modifier = { factor = 1.8 OR = { has_government = democratic has_government = neutrality } }
				set_variable = { flower_ai_stk_style = 2 }
			}
			25 = {
				modifier = { factor = 1.5 num_of_factories > 55 }

			}
		}
	}

	#1. 先按真实行情执行空头爆仓强平、重算浮盈、检查【一代股神】与【天台排队中】里程碑
