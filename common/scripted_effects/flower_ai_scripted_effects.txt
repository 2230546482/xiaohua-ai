		limit = { check_variable = { ubs_save < 0 } }

#从抽卡队列中自动弹出下一轮待选界面（彻底杜绝额外抽卡与定时抽卡互相挤掉）
flower_ai_pop_next_queued_draft = {
	if = {
		limit = { is_ai = no }
		set_variable = { flower_draft_gui_mode = 0 }
		if = {
			limit = { check_variable = { flower_queue_starter > 0 } }
			subtract_from_variable = { flower_queue_starter = 1 }
			set_country_flag = flower_draft_manual_open
			flower_ai_open_starter_selection = yes
		}
		else_if = {
			limit = {
				check_variable = { flower_queue_branch > 0 }
				flower_ai_has_unbranched_archetype = yes
			}
			subtract_from_variable = { flower_queue_branch = 1 }
			set_country_flag = flower_draft_manual_open
			flower_ai_open_branch_evolution_event = yes
		}
		else_if = {
			limit = { check_variable = { flower_queue_hex > 0 } }
			subtract_from_variable = { flower_queue_hex = 1 }
			set_country_flag = flower_draft_manual_open
			flower_ai_open_roguelike_draft = yes
		}
		else = {
			if = {
				limit = { check_variable = { flower_queue_branch > 0 } }
				set_variable = { flower_queue_branch = 0 }
			}
			clr_country_flag = flower_roguelike_gui_open
			flower_ai_recalc_pending_drafts = yes
		}
	}
}

#重算开局畸变投票全员就绪状态（必须全部人类玩家就绪后方可生效，支持单人秒开与多人随时取消投票）
flower_ai_recalc_anomaly_ready = {
	set_variable = { global.flower_human_total = 0 }
	set_variable = { global.flower_human_ready = 0 }
	set_variable = { global.flower_human_skip = 0 }
	every_country = {
		limit = { is_ai = no }

		if = {

			add_to_variable = { global.flower_human_ready = 1 }

		if = {
			limit = { has_country_flag = flower_anomaly_player_skip }
			add_to_variable = { global.flower_human_skip = 1 }
		}
	}
	if = {
		limit = {
			NOT = { has_global_flag = flower_anomaly_voting_settled }
			check_variable = { global.flower_human_total > 0 }
			NOT = { check_variable = { global.flower_human_ready < global.flower_human_total } }
		}
		if = {
			limit = {
				NOT = { check_variable = { global.flower_human_skip < global.flower_human_total } }
			}
			clr_global_flag = flower_anomaly_voting_open
			set_global_flag = flower_anomaly_voting_settled

				limit = { is_ai = no }
				clr_country_flag = flower_anomaly_panel_open
				clr_country_flag = flower_anomaly_player_ready
				clr_country_flag = flower_anomaly_player_skip
			}
		}
		else = {
			flower_ai_lock_in_anomalies_and_start = yes
		}
	}
}
