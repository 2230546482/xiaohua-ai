#弹出下一轮待选界面
flower_ai_pop_next_queued_draft = {
	if = {
		limit = { is_ai = no }
		set_variable = { flower_draft_gui_mode = 0 }
		if = {
			limit = { check_variable = { flower_queue_starter > 0 } }
			subtract_from_variable = { flower_queue_starter = 1 }
			set_country_flag = flower_draft_manual_open
			flower_ai_open_starter_selection = yes
		}
		else_if = {
			limit = {
				check_variable = { flower_queue_branch > 0 }
				flower_ai_has_unbranched_archetype = yes
			}
			subtract_from_variable = { flower_queue_branch = 1 }
			set_country_flag = flower_draft_manual_open
			flower_ai_open_branch_evolution_event = yes
		}
		else_if = {
			limit = { check_variable = { flower_queue_hex > 0 } }
			subtract_from_variable = { flower_queue_hex = 1 }
			set_country_flag = flower_draft_manual_open
			flower_ai_open_roguelike_draft = yes
		}
		else = {
			if = {
				limit = { check_variable = { flower_queue_branch > 0 } }
				set_variable = { flower_queue_branch = 0 }
			}
			clr_country_flag = flower_roguelike_gui_open
			flower_ai_recalc_pending_drafts = yes
		}
	}
}

#超级骨头专属：对宣战的主要国家执行小规模超位魔法核爆（调用统一多维战术智能选点，精准锁定重兵与工业核心）
flower_ai_gutou_super_tier_magic_nuke = {
	if = {
		limit = {
			DVL = {
				exists = yes
				OR = {
					has_idea = red_super_gutou
					has_idea = red_super_gutou_phase2
				}
			}
			var:flower_gutou_nuke_target = {
				exists = yes
				is_major = yes
				NOT = { tag = DVL }
				NOT = { has_country_flag = flower_gutou_super_magic_nuked }
			}
		}
		var:flower_gutou_nuke_target = {
			set_country_flag = { flag = flower_gutou_super_magic_nuked days = 730 value = 1 }
			set_variable = { flower_dvl_desp_mode = 0 }
			#第一发：战略中枢或前线重兵矛头
			set_variable = { flower_dvl_slot_mode = 1 }
			flower_ai_dvl_pick_best_state = yes
			if = {
				limit = { check_variable = { flower_dvl_best_st_id > 0 } }
				var:flower_dvl_best_st_id = { set_state_flag = flower_dvl_target_tmp }
				set_temp_variable = { flower_gutou_nuke_st = flower_dvl_best_st_id }
				DVL = {
					launch_nuke = {
						state = var:flower_gutou_nuke_st
						use_nuke = no
					}
				}
			}
			#第二发：军工心脏、大机场或核设施
			set_variable = { flower_dvl_slot_mode = 2 }
			flower_ai_dvl_pick_best_state = yes
			if = {
				limit = { check_variable = { flower_dvl_best_st_id > 0 } }
				var:flower_dvl_best_st_id = { set_state_flag = flower_dvl_target_tmp }
				set_temp_variable = { flower_gutou_nuke_st = flower_dvl_best_st_id }
				DVL = {
					launch_nuke = {
						state = var:flower_gutou_nuke_st
						use_nuke = no
					}
				}
			}
			#第三发：前线补给枢纽或次高价值重兵集群
