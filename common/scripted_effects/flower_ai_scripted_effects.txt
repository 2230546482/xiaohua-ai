#检测是否达成【双流派大成/双修共鸣】
flower_ai_check_dual_mastery_synergy = {
	if = {
		limit = {
			NOT = { has_country_flag = flower_dual_synergy_unlocked }
		}
		set_temp_variable = { mastered_cnt = 0 }
		if = { limit = { has_idea = flower_arty_stage_3_idea } add_to_temp_variable = { mastered_cnt = 1 } }
		if = { limit = { has_idea = flower_tank_stage_3_idea } add_to_temp_variable = { mastered_cnt = 1 } }
		if = { limit = { has_idea = flower_corp_stage_3_idea } add_to_temp_variable = { mastered_cnt = 1 } }
		if = { limit = { has_idea = flower_air_stage_3_idea } add_to_temp_variable = { mastered_cnt = 1 } }
		if = { limit = { has_idea = flower_wave_stage_3_idea } add_to_temp_variable = { mastered_cnt = 1 } }
		if = { limit = { has_idea = flower_magic_stage_3_idea } add_to_temp_variable = { mastered_cnt = 1 } }
		if = { limit = { has_idea = flower_nightingale_stage_3_idea } add_to_temp_variable = { mastered_cnt = 1 } }
		if = { limit = { has_idea = flower_guerrilla_stage_3_idea } add_to_temp_variable = { mastered_cnt = 1 } }
		if = { limit = { has_idea = flower_undead_stage_3_idea } add_to_temp_variable = { mastered_cnt = 1 } }
		if = { limit = { has_idea = flower_steampunk_stage_3_idea } add_to_temp_variable = { mastered_cnt = 1 } }
		if = { limit = { check_variable = { flower_slayer_stacks > 5 } } add_to_temp_variable = { mastered_cnt = 1 } }
		if = { limit = { check_variable = { flower_builder_stacks > 5 } } add_to_temp_variable = { mastered_cnt = 1 } }
		if = {
			limit = {
				has_idea = flower_mystery_egg_hatched_idea
				has_idea = flower_blood_wager_victory_idea
			}
			add_to_temp_variable = { mastered_cnt = 1 }
		}
		if = {
			limit = { check_variable = { mastered_cnt > 1 } }
			set_country_flag = flower_dual_synergy_unlocked
			country_event = { id = FLWAI.26 days = 1 }
		}
	}
}

#《记仇小本本》（战地拾荒扒衣流）叠层重算与缴获爆装（属性上限10层，强化装备缴获与补给自持）
flower_ai_recalc_slayer_book = {
	if = {
		limit = {
			NOT = { has_dynamic_modifier = { modifier = flower_slayer_book_dynamic_mod } }
		}
		add_dynamic_modifier = { modifier = flower_slayer_book_dynamic_mod }
	}
	set_temp_variable = { slayer_eff = flower_slayer_stacks }
	if = {
		limit = { check_variable = { slayer_eff > 10 } }
		set_temp_variable = { slayer_eff = 10 }
	}
	set_variable = { flower_slayer_atk_var = slayer_eff }
	multiply_variable = { flower_slayer_atk_var = 0.025 }
	set_variable = { flower_slayer_brk_var = slayer_eff }
	multiply_variable = { flower_slayer_brk_var = 0.025 }
	set_variable = { flower_slayer_cap_var = slayer_eff }
	multiply_variable = { flower_slayer_cap_var = 0.08 }
	set_variable = { flower_slayer_sup_var = slayer_eff }
	multiply_variable = { flower_slayer_sup_var = -0.025 }
	set_variable = { flower_slayer_mp_var = slayer_eff }
	multiply_variable = { flower_slayer_mp_var = 800 }
	if = {
		limit = {
			check_variable = { flower_slayer_stacks > 4 }
			NOT = { has_country_flag = flower_slayer_milestone_5_done }
		}
		set_country_flag = flower_slayer_milestone_5_done
		country_event = { id = FLWAI.21 days = 1 }
	}
	if = {
		limit = {
			check_variable = { flower_slayer_stacks > 9 }
			NOT = { has_country_flag = flower_slayer_milestone_10_done }
		}
		set_country_flag = flower_slayer_milestone_10_done
		country_event = { id = FLWAI.21 days = 1 }
	}
	flower_ai_check_dual_mastery_synergy = yes
}

flower_ai_add_slayer_stack = {
	add_to_variable = { flower_slayer_stacks = 1 }
	#战地拾荒：每叠1层当场从战场扒回一批军火战利品
	add_equipment_to_stockpile = {
		type = infantry_equipment
		amount = 2000
	}
	add_equipment_to_stockpile = {
		type = artillery_equipment
		amount = 250
	}
	add_equipment_to_stockpile = {
		type = support_equipment
		amount = 150
	}
	#分叉《拆家专业户》：每次叠层战利品翻倍再白嫖资金
	if = {
		limit = { has_idea = flower_evo_slayer_dismantle_idea }
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = 2000
		}
		add_equipment_to_stockpile = {
			type = artillery_equipment
			amount = 250
		}
		add_equipment_to_stockpile = {
			type = support_equipment
			amount = 150
		}
		add_to_variable = { another_country_currency_reserve = 150 }
	}
	#满5层后每次叠层还会连地皮带机器拆回1座免费军工
	if = {
		limit = { has_country_flag = flower_slayer_milestone_5_done }
		random_owned_controlled_state = {
			limit = {
				free_building_slots = {
					building = arms_factory
					size > 0
					include_locked = yes
				}
			}
			add_extra_state_shared_building_slots = 1
			add_building_construction = {
				type = arms_factory
				level = 1
				instant_build = yes
			}
		}
	}
	flower_ai_recalc_slayer_book = yes
}
