    on_startup = {
        effect = {
            if = {
                limit = {
                    has_game_rule = {
                        rule = FLOWER_AI_SUPER_BOSS_RULE
                        option = SUPER_DVL
                    }
                }
                DVL = {
                    if = {
                        limit = { is_ai = yes }
                        add_ideas = red_super_gutou
                    }
                }
            }
            else_if = {
                limit = {
                    has_game_rule = {
                        rule = FLOWER_AI_SUPER_BOSS_RULE
                        option = SUPER_SOV_DVL
                    }
                }
                DVL = {
                    if = {
                        limit = { is_ai = yes }
                        add_ideas = red_super_gutou
                    }
                }
                SOV = {
                    if = {
                        limit = { is_ai = yes }
                        set_technology = {
                            improved_heavy_tank_chassis = 1
                            antitank1 = 1
                            antitank2 = 1
                            antitank3 = 1
                        }
                    }
                }
            }

            if = {
                limit = {
                    has_game_rule = {
                        rule = FLOWER_AI_DIFFICULTY_RULE
                        option = APOCALYPSE
                    }
                }
                set_global_flag = red_apocalypse_ai
                every_country = {
                    limit = { is_ai = no }
                    set_country_flag = hua_ai_player
                    add_ideas = red_apocalypse_player
                }
                every_country = {
                    limit = {
                        is_ai = yes
                        NOT = { tag = ZDW }
                        NOT = { tag = WL1 }
                    }
                    add_ideas = red_apocalypse_ai
                }
            }
            else_if = {
                limit = {
                    has_game_rule = {
                        rule = FLOWER_AI_DIFFICULTY_RULE
                        option = MOST_HARD
                    }
                }
                set_global_flag = red_most_hard
                every_country = {
                    limit = { is_ai = no }
                    set_country_flag = hua_ai_player
                    add_ideas = red_most_hard_player
                }
                every_country = {
                    limit = {
                        is_ai = yes
                        NOT = { tag = ZDW }
                        NOT = { tag = WL1 }
                    }
                    add_ideas = red_most_hard_ai
                }
            }
            else_if = {
                limit = {
                    has_game_rule = {
                        rule = FLOWER_AI_DIFFICULTY_RULE
                        option = HARD
                    }
                }
                set_global_flag = red_hard
                every_country = {
                    limit = { is_ai = no }
                    set_country_flag = hua_ai_player
                }
                every_country = {
                    limit = {
                        is_ai = yes
                        NOT = { tag = ZDW }
                        NOT = { tag = WL1 }
                    }
                    add_ideas = red_hard_ai
                }
            }
            else_if = {
                limit = {
                    has_game_rule = {
                        rule = FLOWER_AI_DIFFICULTY_RULE
                        option = NORMAL_HARD
                    }
                }
                set_global_flag = red_normal_hard
                every_country = {
                    limit = { is_ai = no }
                    set_country_flag = hua_ai_player
                }
                every_country = {
                    limit = {
                        is_ai = yes
                        NOT = { tag = ZDW }
                        NOT = { tag = WL1 }
                    }
                    add_ideas = red_normal_hard_ai
                }
            }
            else_if = {
                limit = {
                    has_game_rule = {
                        rule = FLOWER_AI_DIFFICULTY_RULE
                        option = LOGIC_ONLY
                    }
                }
                set_global_flag = red_none_hard
                set_global_flag = flower_anomaly_voting_open
                every_country = {
                    limit = { is_ai = no }
                    set_country_flag = hua_ai_player
                }
            }
            else = {
                random_country = {
                    limit = {
                        is_ai = no
                    }
                    country_event = FLWAI.1
                }
            }
            if = {
                limit = {
                    NOT = {
                        has_game_rule = {
                            rule = FLOWER_AI_DIFFICULTY_RULE
                            option = DEFAULT_EVENT
                        }
                    }
                }
                set_global_flag = flower_anomaly_voting_open

            flower_ai_recalc_anomaly_ready = yes

            every_country = {
                set_variable = { flower_cached_factories = num_of_factories }
                if = {
                    limit = { is_major = yes }
                    set_country_flag = flower_cached_is_major
                }
            }
        }
    }
    #月度补挂难度精神与肉鸽成长推进
    on_monthly = {
        effect = {
            #投票阶段断线与变动防卡死自动重算
            if = {
                limit = { has_global_flag = flower_anomaly_voting_open }
                flower_ai_recalc_anomaly_ready = yes
            }
            #【老实赚钱很枯燥吧】实盘全球股市月度跳价与分红（全局每月仅触发一次）
            if = {
                limit = {
                    has_global_flag = flower_anomaly_3_active
                    NOT = { has_global_flag = flower_stk_monthly_cd }
                }
                set_global_flag = { flag = flower_stk_monthly_cd days = 25 value = 1 }
                flower_stock_monthly_tick = yes
            }
            #缓存各国非投降状态下的战前/峰值工厂数与主要国家身份（防止投降或吞并瞬间数值归零导致误判）
            if = {
                limit = {
                    has_capitulated = no
                    check_variable = { num_owned_states > 0 }
                }
                if = {
                    limit = {
                        OR = {
                            has_war = no
                            check_variable = { num_of_factories > flower_cached_factories }
                        }
                    }
                    set_variable = { flower_cached_factories = num_of_factories }
                }
                if = {
                    limit = { is_major = yes }
                    set_country_flag = flower_cached_is_major
                }
                else_if = {
                    limit = { has_war = no }
                    clr_country_flag = flower_cached_is_major
                }
            }
            #《基建狂魔的流水账》造厂达标自动叠层返利
            if = {
                limit = {
                    check_variable = { flower_builder_stacks > 0 }
                    check_variable = { num_of_factories > flower_builder_next_goal }
                }
                add_to_variable = { flower_builder_next_goal = 15 }
                flower_ai_add_builder_stack = yes
            }
            #《记仇小本本》战时激战满4个月自动叠层
            if = {
                limit = {
                    check_variable = { flower_slayer_stacks > 0 }
                    has_war = yes
                }
                add_to_variable = { flower_slayer_war_months = 1 }
                if = {
                    limit = { check_variable = { flower_slayer_war_months > 3 } }
                    set_variable = { flower_slayer_war_months = 0 }
                    flower_ai_add_slayer_stack = yes
                }
            }
            #12大流派专属局内成长里程碑月度检测（玩家与AI通用）
            flower_ai_check_archetype_milestones = yes

            if = {
                limit = {
                    OR = {
                        has_idea = flower_undead_stage_1_idea
                        has_idea = flower_undead_stage_2_idea
                        has_idea = flower_undead_stage_3_idea
                        has_country_flag = flower_card_owned_27
                    }
                }

                    type = wangling_equipment_0
                    amount = 1500
                }
                if = {
                    limit = {
                        OR = {
                            has_idea = flower_evo_undead_plague_idea
                            has_idea = flower_combo_undead_sweatshop_idea
                        }
                    }
                    add_equipment_to_stockpile = {
                        type = wangling_equipment_0
                        amount = 1500
                    }
                }
                if = {
                    limit = {
                        has_war = yes
                        OR = {
                            has_idea = flower_undead_stage_2_idea
                            has_idea = flower_undead_stage_3_idea
                            has_idea = flower_evo_undead_plague_idea
                        }
                    }
                    add_to_variable = { flower_undead_war_months = 1 }
                    if = {
                        limit = {
                            OR = {
                                AND = {
                                    has_idea = flower_evo_undead_plague_idea
                                    check_variable = { flower_undead_war_months > 1 }
                                }
                                check_variable = { flower_undead_war_months > 2 }
                            }
                        }
                        set_variable = { flower_undead_war_months = 0 }
                        flower_ai_summon_undead_wave = yes
                    }
                }
            }
            #流派八《野火游击队》战时自动拉起野火游击纵队（人海分叉缩至每2个月）

            #【华尔街军火大亨】年度金融风暴事件

                limit = {
                    has_global_flag = flower_anomaly_3_active
                    OR = {
                        is_ai = no

                    }
                }

                    limit = {
