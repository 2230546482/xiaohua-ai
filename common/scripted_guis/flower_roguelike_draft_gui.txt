						num_of_factories < 50
						has_equipment = { infantry_equipment > 1999 }
						has_equipment = { artillery_equipment > 199 }
					}
					add_equipment_to_stockpile = {
						type = infantry_equipment
						amount = -2000
					}
					add_equipment_to_stockpile = {
						type = artillery_equipment
						amount = -200
					}
				}
				else_if = {
					limit = {
						num_of_factories > 49
						has_equipment = { infantry_equipment > 4999 }
						has_equipment = { artillery_equipment > 499 }
					}
					add_equipment_to_stockpile = {
						type = infantry_equipment
						amount = -5000
					}
					add_equipment_to_stockpile = {
						type = artillery_equipment
						amount = -500
					}
				}
				else = {
					add_to_variable = { ubs_save = -300 }
					flower_ai_sync_hcp_ui = yes
				}
				add_war_support = -0.05
				flower_ai_roll_roguelike_draft = yes
			}
			flower_draft_peek_map_btn_click = {
				clr_country_flag = flower_roguelike_gui_open
			}
			flower_draft_close_btn_click = {
				clr_country_flag = flower_roguelike_gui_open
			}
			flower_draft_cat_calm_btn_click = {
				set_country_flag = { flag = flower_cat_hissing days = 2 value = 1 }
			}
			flower_draft_cat_hiss_btn_click = {
				clr_country_flag = flower_cat_hissing
			}
		}
	}
