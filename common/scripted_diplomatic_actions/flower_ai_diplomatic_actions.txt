#小花AI外交协作动作
scripted_diplomatic_actions = {
	#拨付200HCP财政援助
	flower_ai_hcp_financial_aid = {
		allowed = {
			ROOT = { is_ai = no }
		}
		visible = {
			ROOT = { is_ai = no }
			FROM = {
				exists = yes
				NOT = { has_war_with = ROOT }
			}
		}
		selectable = {
			ROOT = {
				check_variable = { ubs_save > 199.999 }
			}
			FROM = {
				NOT = { has_war_with = ROOT }
			}
		}
		cost = 15
		requires_acceptance = no
		show_acceptance_on_action_button = no
		send_description = FLOWER_AI_HCP_FINANCIAL_AID_SEND_DESC
		complete_effect = {
			ROOT = {
				HCP_subtract_banknote_200 = yes
			}
			FROM = {
				if = {
					limit = {
						check_variable = { ubs_load > 199.999 }
					}
					subtract_from_variable = { ubs_load = 200 }
				}
				else = {
					HCP_add_banknote_200 = yes
				}
			}
			AHM_economy_request_refresh = yes
			FROM = {
				if = {
					limit = { is_ai = no }
					set_temp_variable = { temp_alert_id = 902 }
					set_temp_variable = { temp_tag = ROOT }
					set_temp_variable = { temp_tag2 = FROM }
					flower_ai_push_bottom_right_alert = yes
				}
			}
		}
		ai_desire = {
			base = 0
		}
	}

	#单国战线净空协定
	flower_ai_toggle_ally_border_clearance = {
		allowed = {
			ROOT = { is_ai = no }
		}
		visible = {
			ROOT = { is_ai = no }
			FROM = {
				is_ai = yes
				NOT = { has_war_with = ROOT }
				OR = {
					is_in_faction_with = ROOT
					is_subject_of = ROOT
					has_military_access_to = ROOT
				}
			}
		}
		selectable = {
			FROM = {
				is_ai = yes
				NOT = { has_war_with = ROOT }
			}
		}
		cost = 0
		requires_acceptance = no
		show_acceptance_on_action_button = no
		send_description = FLOWER_AI_TOGGLE_ALLY_CLEARANCE_SEND_DESC
		complete_effect = {
			FROM = {
				if = {
					limit = {
						has_country_flag = flower_ai_dont_crowd_@ROOT
					}
					clr_country_flag = flower_ai_dont_crowd_@ROOT
					add_ai_strategy = {
						type = dont_defend_ally_borders
						id = ROOT
						value = -500
					}
					add_ai_strategy = {
						type = force_defend_ally_borders
						id = ROOT
						value = 500
					}
				}
				else = {
					set_country_flag = flower_ai_dont_crowd_@ROOT
					add_ai_strategy = {
						type = dont_defend_ally_borders
						id = ROOT
						value = 500
					}
					add_ai_strategy = {
						type = force_defend_ally_borders
						id = ROOT
						value = -500
					}
				}
			}
		}
		ai_desire = {
			base = 0
		}
	}
}
