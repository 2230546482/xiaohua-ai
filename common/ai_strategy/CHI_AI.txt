#外交与战线策略
CHI_generic_capital_shield = {
    allowed = {
        OR = {
            is_chinese_country = yes
        }
    }
    enable = {
        has_war = yes
        surrender_progress < 0.25
        NOT = {
            any_enemy_country = { surrender_progress > 0.35 }
        }
    }
    abort_when_not_enabled = yes

    ai_strategy = {
        type = front_unit_request
        state_trigger = {
            OR = {
                AND = {
                    is_capital = yes
                    is_controlled_by = FROM.FROM
                }
                AND = {
                    any_neighbor_state = {
                        is_capital = yes
                        is_controlled_by = FROM.FROM
                    }
                    is_controlled_by = FROM.FROM
                }
            }
        }
        value = 80
    }
    ai_strategy = {
        type = front_control
        state_trigger = {
            OR = {
                is_capital = yes
                any_neighbor_state = {
                    is_capital = yes
                    is_controlled_by = FROM.FROM
                }
            }
            is_controlled_by = FROM.FROM
            any_neighbor_state = {
                controller = { has_war_with = FROM.FROM }
            }
        }
        priority = 185
        ordertype = front
        execution_type = careful
        manual_attack = yes
    }
}
