	ROM_war_with_Kivotos = {
		allowed = {
			original_tag = ROM
		}
		enable = {
			any_enemy_country = {
				Kivotos_is_kivotos_country = yes
			}
		}
		abort_when_not_enabled = yes

		ai_strategy = {
			type = front_control
			country_trigger = {
				Kivotos_is_kivotos_country = yes
			}
			ratio = 0.8
			ordertype = front
			execution_type = rush_weak
			manual_attack = no
		}

		ai_strategy = {
			type = front_unit_request
			country_trigger = {
				Kivotos_is_kivotos_country = yes
			}
			value = 100
		}
	}
