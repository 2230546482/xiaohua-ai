flower_counter_player_armor_strat = {
	allowed = { NOT = { tag = ZDW tag = WL1 } }
	enable = {
		OR = { has_global_flag = flower_anomaly_1_active has_global_flag = flower_anomaly_4_active }
		is_ai = yes
		has_war = yes
		num_of_factories > 20
		any_enemy_country = {
			is_ai = no
			has_country_flag = flower_roguelike_tank_ai
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = equipment_production_factor
		id = anti_tank
		value = 180
	}
	ai_strategy = {
		type = equipment_production_min_factories
		id = anti_tank
		value = 4
	}
	ai_strategy = {
		type = equipment_production_factor
		id = artillery
		value = 60
	}
}
