KMK_war_with_DVL = {
	allowed = {
		original_tag = KMK
	}
	enable = {
		has_war_with = DVL
	}
	abort_when_not_enabled = yes
	
	ai_strategy = {
		type = conquer		#征服
		id = "DVL"			
		value = 200
	}
	
	ai_strategy = {
		type = front_unit_request #前线部队
		tag = DVL
		value = 200 #lets put more units, should be safe
	}
	
	
}

KMK_war_with_SOV = {
	allowed = {
		original_tag = KMK
	}
	enable = {
		has_war_with = SOV
		not = {
			has_war_with = GER
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = invade		#入侵
		id = "SOV"
		value = 200
	}
	ai_strategy = {
		type = conquer		#征服
		id = "SOV"
		value = 200
	}
	ai_strategy = {
		type = invasion_unit_request
		tag = SOV
		value = 10
	}
	ai_strategy = {
		type = front_unit_request		#前线部队
		tag = SOV
		value = 200		#lets put more units, should be safe
	}
}
