#三、阶段6~7跨太平洋反攻樱花（JAP）海空两栖协同强化

    enable = {
        OR = {
            has_country_flag = has_PAU_xiaoyao6
            has_country_flag = has_PAU_xiaoyao7
        }
        country_exists = JAP
        NOT = { has_country_flag = PAU_sakura_super_fired }
        NOT = {
            AND = {
                PAO = { is_ai = no }

            }
        }
    }
    abort_when_not_enabled = yes

    ai_strategy = {
        type = naval_invasion_focus

    }

        type = invasion_unit_request

        value = 150

    ai_strategy = {

            is_controlled_by = JAP
